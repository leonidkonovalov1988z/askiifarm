﻿using AsciiFarm.Components;
using AsciiFarm.Content;

public sealed class GameApplication
{
    private readonly GameSettings _settings;

    public GameApplication(GameSettings settings) => _settings = settings;

    // App/GameApplication.cs — только метод Run(), остальное без изменений
    public void Run()
    {
        if (Console.IsInputRedirected)
            throw new InvalidOperationException("Нужна настоящая консоль: ввод перенаправлен.");

        var world = GameWorldBuilder.Build(_settings);
        var channel = new CommandChannel(_settings.CommandQueueCapacity);
        var frames = new FrameExchange<RenderFrame>(
            () => new RenderFrame(_settings.ScreenWidth, _settings.ScreenHeight));

        using var console = new DoubleBufferedConsole(_settings.ScreenWidth, _settings.ScreenHeight);
        using var shutdown = new CancellationTokenSource();
        var errors = new ConcurrentQueue<Exception>();

        // Загружаем привязки из того же источника, что и остальной контент
        KeyBindings bindings;
        if (_settings.UseJsonContent && Directory.Exists(_settings.ContentRoot))
        {
            bindings = new ContentLoader(_settings.ContentRoot).LoadKeyBindings();
        }
        else
        {
            bindings = GameContent.CreateKeyBindings();
        }

        var simulation = new SimulationLoop(world, channel, SystemPipeline.CreateSimulationSystems(),
            new SnapshotSystem(frames), frames, shutdown);
        // В методе Run(), при создании ConsolePresentSystem:
        var fpsCounter = world.Resource<FpsCounter>();  // ДОБАВИТЬ
        var render = new RenderLoop(console, frames,
            new ConsolePresentSystem(console, frames, fpsCounter), 1000 / _settings.RenderFps);
        var input = new InputLoop(channel, bindings, _settings.InputPollMs);

        var threads = new[]
        {
        Start("Simulation", () => simulation.Run(shutdown.Token), shutdown, errors),
        Start("Render",     () => render.Run(shutdown.Token),     shutdown, errors),
        Start("Input",      () => input.Run(shutdown.Token),      shutdown, errors),
    };

        foreach (var thread in threads) thread.Join();
        channel.Close();
        if (!errors.IsEmpty) throw new AggregateException("Сбой игрового потока.", errors);

        world.Guard.Detach();
        var state = world.Resource<GameState>();
        if (state.NeedsSave) SaveGame(world);
    }
    // Заменить метод SaveGame на:

    private void SaveGame(World world)
    {
        var registry = ComponentSerializerRegistry.AutoRegister(
            System.Reflection.Assembly.GetExecutingAssembly());
        var saveSystem = new ComponentSaveSystem(registry);
        saveSystem.Save(_settings.MapPath, world);
    }

    private static Thread Start(string name, Action body,
                                CancellationTokenSource shutdown, ConcurrentQueue<Exception> errors)
    {
        var thread = new Thread(() =>
        {
            try { body(); }
            catch (Exception ex) { errors.Enqueue(ex); shutdown.Cancel(); }
        })
        { Name = name, IsBackground = true };
        thread.Start();
        return thread;
    }
}.

﻿using AsciiFarm.Components;
using AsciiFarm.Content;

using askiifarm.Serialization;
using System.Reflection;

public static class GameWorldBuilder
{
    public static World Build(GameSettings settings)
    {
        var world = new World();

        // Создаём реестр сериализаторов (авто-регистрация из сборки)
        var registry = ComponentSerializerRegistry.AutoRegister(Assembly.GetExecutingAssembly());
        var saveSystem = new ComponentSaveSystem(registry);

        // Загружаем мир из файла сохранения
        if (ComponentSaveSystem.IsNewFormat(settings.MapPath))
        {
            saveSystem.Load(settings.MapPath, world);
        }
        else if (File.Exists(settings.MapPath))
        {
            // Legacy формат — загружаем и конвертируем
            var legacyStorage = new GameStorage();
            var legacyData = legacyStorage.LoadLatest<GameDataV0_0_3>(settings.MapPath);

            var map = world.CreateEntity();
            world.Add(map, new TileMapComponent { Tiles = legacyData.Map });

            var player = world.CreateEntity();
            world.Add(player, new Position(legacyData.PlayerX, legacyData.PlayerY));
            world.Add(player, new PlayerTag());
            world.Add(player, new Glyph
            {
                Symbol = '☺',
                Foreground = ConsoleColor.Yellow,
                Background = ConsoleColor.Black,
                Layer = 10
            });

            // Сохраняем в новом формате
            saveSystem.Save(settings.MapPath, world);
        }
        else
        {
            // Файла нет — создаём мир по умолчанию
            // (здесь можно загрузить map_raw.txt или создать процедурно)
            throw new FileNotFoundException($"Файл сохранения не найден: {settings.MapPath}");
        }

        // Загружаем ресурсы (не сохраняются, всегда из JSON или хардкода)
        TilePalette palette;
        CharacterPalette charPalette;

        if (settings.UseJsonContent && Directory.Exists(settings.ContentRoot))
        {
            (palette, _, charPalette) = GameContent.LoadFromFiles(settings.ContentRoot);
        }
        else
        {
            palette = GameContent.CreatePalette();
            charPalette = GameContent.CreateCharacterPalette();
        }

        // В методе Build, после инициализации других ресурсов:

        // В методе Build, среди прочих SetResource:
        var weatherData = WeatherCsvLoader.Load(settings.WeatherCsvPath, out var weatherError);
        world.SetResource(new WeatherDataSource
        {
            HourlyData = weatherData,
            LoadError = weatherError
        });
        world.SetResource(new WeatherCurrent());

        var gameState = new GameState
        {
            GameStartDate = DateTime.Now,
            TimeSpeed = 1.0f,
            MaxTimeStep = settings.MaxTimeStep
        };
        

        world.SetResource(palette);
        world.SetResource(charPalette);
        world.SetResource(gameState);
        world.SetResource(new CommandBuffer());
        world.SetResource(new FpsCounter());
        world.SetResource(new FpsLimiter(settings.SimulationFps));
        world.SetResource(settings);
        return world;
    }
}.

using AsciiFarm.Rendering;
using AsciiFarm.Systems;
using askiifarm;
using askiifarm.Rendering;
namespace AsciiFarm.App;

public sealed class RenderLoop
{
    private readonly DoubleBufferedConsole _console;
    private readonly FrameExchange<RenderFrame> _frames;
    private readonly ConsolePresentSystem _present;
    private readonly int _frameMs;

    public RenderLoop(DoubleBufferedConsole console, FrameExchange<RenderFrame> frames,
                      ConsolePresentSystem present, int frameMs)
    {
        _console = console; _frames = frames; _present = present; _frameMs = frameMs;
    }

    public void Run(CancellationToken token)
    {
        _console.AttachToCurrentThread();       // консоль — только наша
        _frames.AttachReader();

        while (!token.IsCancellationRequested)
        {
            _present.Present();
            Thread.Sleep(_frameMs);
        }
    }
}.

﻿// App/SimulationLoop.cs
using AsciiFarm.Core;
using AsciiFarm.Data;
using AsciiFarm.Input;
using AsciiFarm.Rendering;
using AsciiFarm.Systems;
namespace AsciiFarm.App;

public sealed class SimulationLoop
{
    private readonly World _world;
    private readonly CommandChannel _channel;
    private readonly ISystem[] _systems;
    private readonly SnapshotSystem _snapshot;
    private readonly FrameExchange<RenderFrame> _frames;
    private readonly CancellationTokenSource _shutdown;

    public SimulationLoop(World world, CommandChannel channel, ISystem[] systems,
                          SnapshotSystem snapshot, FrameExchange<RenderFrame> frames,
                          CancellationTokenSource shutdown)
    {
        _world = world; _channel = channel; _systems = systems;
        _snapshot = snapshot; _frames = frames; _shutdown = shutdown;
    }

    public void Run(CancellationToken token)
    {
        _world.Guard.AttachToCurrentThread();
        _frames.AttachWriter();

        var state = _world.Resource<GameState>();
        var buffer = _world.Resource<CommandBuffer>();
        var limiter = _world.Resource<FpsLimiter>();

        while (!token.IsCancellationRequested && state.IsRunning)
        {
            // Промотка времени: используем MaxTimeStep
            if (state.FastForwardUntil.HasValue)
            {
                double remaining = state.FastForwardUntil.Value - state.ElapsedTime;
                double step = Math.Min(remaining, state.MaxTimeStep);

                state.ElapsedTime += step;
                state.GameTimeElapsed += step;
                state.Tick++;

                using (_world.Guard.Enter())
                {
                    _channel.DrainTo(buffer.Items);
                    foreach (var system in _systems) system.Update(_world);
                    _snapshot.Update(_world);
                }

                if (state.ElapsedTime >= state.FastForwardUntil.Value)
                    state.FastForwardUntil = null;

                continue;
            }

            // Обычный режим с ограничением FPS
            while (!limiter.BeginTick())
            {
                Thread.Sleep(1);
                if (token.IsCancellationRequested || !state.IsRunning) break;
            }
            if (token.IsCancellationRequested || !state.IsRunning) break;

            using (_world.Guard.Enter())
            {
                _channel.DrainTo(buffer.Items);
                foreach (var system in _systems) system.Update(_world);
                _snapshot.Update(_world);
            }
        }

        _shutdown.Cancel();
    }
}.

using System;
using System.Collections.Generic;
using System.IO;
using System.Text.Json;
using System.Text.Json.Serialization;
using AsciiFarm.Components;
using AsciiFarm.Data;
using AsciiFarm.Input;
using askiifarm;

namespace AsciiFarm.Content;

/// <summary>
/// Загружает контент из JSON-файлов. Правка баланса/цветов/клавиш — без перекомпиляции.
/// </summary>
public sealed class ContentLoader
{
    private static readonly JsonSerializerOptions JsonOptions = new()
    {
        PropertyNameCaseInsensitive = true,
        Converters = { new JsonStringEnumConverter() },
        WriteIndented = true
    };

    // DTO для палитры
    private sealed class PaletteDto
    {
        public TileStyleDto Fallback { get; set; } = new();
        public Dictionary<string, TileStyleDto> Tiles { get; set; } = new();
    }
    private sealed class TileStyleDto
    {
        public ConsoleColor Foreground { get; set; } = ConsoleColor.Gray;
        public ConsoleColor Background { get; set; } = ConsoleColor.Black;
        public bool BlocksMovement { get; set; }
    }

    // DTO для привязок
    private sealed class KeyBindingsDto
    {
        public Dictionary<string, string> Bindings { get; set; } = new();
    }

    // DTO для персонажей
    private sealed class CharactersDto
    {
        public string DefaultPlayerKey { get; set; } = "player";
        public Dictionary<string, GlyphDto> Characters { get; set; } = new();
    }
    private sealed class GlyphDto
    {
        public string Symbol { get; set; } = "?";
        public ConsoleColor Foreground { get; set; } = ConsoleColor.White;
        public ConsoleColor Background { get; set; } = ConsoleColor.Black;
        public int Layer { get; set; }
    }

    public string ContentRoot { get; }

    public ContentLoader(string contentRoot)
    {
        ContentRoot = contentRoot;
    }

    public TilePalette LoadPalette(string fileName = "palette.json")
    {
        var path = Path.Combine(ContentRoot, fileName);
        var dto = JsonSerializer.Deserialize<PaletteDto>(File.ReadAllText(path), JsonOptions)
            ?? throw new InvalidOperationException($"Не удалось прочитать {path}");

        var palette = new TilePalette
        {
            Fallback = ToStyle(dto.Fallback)
        };
        foreach (var kvp in dto.Tiles)
        {
            if (kvp.Key.Length == 1)
                palette.Tiles[kvp.Key[0]] = ToStyle(kvp.Value);
        }
        return palette;
    }

    public KeyBindings LoadKeyBindings(string fileName = "keybindings.json")
    {
        var path = Path.Combine(ContentRoot, fileName);
        var dto = JsonSerializer.Deserialize<KeyBindingsDto>(File.ReadAllText(path), JsonOptions)
            ?? throw new InvalidOperationException($"Не удалось прочитать {path}");

        var bindings = new KeyBindings();
        foreach (var kvp in dto.Bindings)
        {
            if (Enum.TryParse<ConsoleKey>(kvp.Key, true, out var key) &&
                Enum.TryParse<GameCommand>(kvp.Value, true, out var cmd))
            {
                bindings.Map[key] = cmd;
            }
        }
        return bindings;
    }

    public CharacterPalette LoadCharacterPalette(string fileName = "characters.json")
    {
        var path = Path.Combine(ContentRoot, fileName);
        var dto = JsonSerializer.Deserialize<CharactersDto>(File.ReadAllText(path), JsonOptions)
            ?? throw new InvalidOperationException($"Не удалось прочитать {path}");

        var palette = new CharacterPalette { DefaultPlayerKey = dto.DefaultPlayerKey };
        foreach (var kvp in dto.Characters)
        {
            palette.Characters[kvp.Key] = new Glyph
            {
                Symbol = kvp.Value.Symbol.Length > 0 ? kvp.Value.Symbol[0] : '?',
                Foreground = kvp.Value.Foreground,
                Background = kvp.Value.Background,
                Layer = kvp.Value.Layer
            };
        }
        return palette;
    }

    private static TileStyle ToStyle(TileStyleDto dto) =>
        new(dto.Foreground, dto.Background, dto.BlocksMovement);
}.

using AsciiFarm.Components;
using AsciiFarm.Data;
using AsciiFarm.Input;
using askiifarm;
namespace AsciiFarm.Content;

/// <summary>
/// Тонкая обёртка над ContentLoader. Позволяет создавать контент из кода (тесты, дефолты)
/// ИЛИ из JSON-файлов (продакшн).
/// </summary>
public static class GameContent
{
    /// <summary>Загрузить всё из JSON (рекомендуемый путь).</summary>
    public static (TilePalette Palette, KeyBindings Keys, CharacterPalette Chars)
        LoadFromFiles(string contentRoot)
    {
        var loader = new ContentLoader(contentRoot);
        return (loader.LoadPalette(), loader.LoadKeyBindings(), loader.LoadCharacterPalette());
    }

    /// <summary>Хардкод-дефолт. Используется, если JSON недоступен.</summary>
    public static TilePalette CreatePalette()
    {
        var p = new TilePalette { Fallback = new TileStyle(ConsoleColor.Gray, ConsoleColor.Black, false) };
        void Set(string chars, ConsoleColor fg, ConsoleColor bg = ConsoleColor.Black, bool blocks = false)
        {
            foreach (char c in chars) p.Tiles[c] = new TileStyle(fg, bg, blocks);
        }
        Set(".,'`;", ConsoleColor.Green);
        Set("\"", ConsoleColor.Yellow);
        Set("♣♠", ConsoleColor.DarkGreen, blocks: true);
        Set("≈", ConsoleColor.Cyan);
        Set("┌┐└┘─│┬┴├┤▒#", ConsoleColor.DarkYellow, blocks: true);
        Set("┼╪", ConsoleColor.DarkYellow);
        Set("Θ", ConsoleColor.Red, blocks: true);
        Set("≡", ConsoleColor.White, ConsoleColor.DarkBlue, blocks: true);
        Set("╔╗╚╝═║╠╣╦╩╬", ConsoleColor.Gray, blocks: true);
        return p;
    }

    public static KeyBindings CreateKeyBindings()
    {
        var b = new KeyBindings();

using System;

using System.Globalization;
using System.IO;

/// <summary>
/// Парсит CSV погоды. Поддерживает два формата:
///   1) Плоский: заголовки в 1-й строке, имена могут содержать единицы
///      ("temperature_2m (°C)").
///   2) Open-Meteo API: заголовки в строке "hourly", затем "hourly_units", затем данные.
/// </summary>
public static class WeatherCsvLoader
{
    private const float WiltingPoint = 0.10f;
    private const float FieldCapacity = 0.38f;
    private const float SnowCorrectionFactor = 0.5f;
    private const float SnowInsulationFactor = 0.15f;
    private const float SnowMeltTempThreshold = 3f;

    public static List<WeatherData> Load(string filePath)
        => Load(filePath, out _);

    // Content/WeatherCsvLoader.cs — заменить метод Load
    public static List<WeatherData> Load(string filePath, out string? error)
    {
        error = null;
        var result = new List<WeatherData>();
        if (!File.Exists(filePath)) { error = $"Файл не найден: {filePath}"; return result; }

        string[] lines;
        try { lines = File.ReadAllLines(filePath); }
        catch (Exception ex) { error = $"Не удалось прочитать: {ex.Message}"; return result; }

        if (lines.Length < 2) { error = "Файл слишком короткий"; return result; }

        int headerIndex = -1;
        for (int i = 0; i < Math.Min(10, lines.Length); i++)
        {
            var lower = lines[i].ToLowerInvariant();
            if (lower.Contains("time") && lower.Contains("temperature"))
            { headerIndex = i; break; }
        }
        if (headerIndex < 0) { error = "Не найдена строка заголовков"; return result; }

        var headers = ParseCsvLine(lines[headerIndex]).Select(NormalizeHeader).ToList();
        int idxTime = headers.IndexOf("time");
        int idxTemp = headers.IndexOf("temperature_2m");
        int idxHum = headers.IndexOf("relative_humidity_2m");
        int idxRad = headers.IndexOf("shortwave_radiation");
        int idxSoil = headers.IndexOf("soil_moisture_7_to_28cm");
        int idxSnow = headers.IndexOf("snow_depth");
        int idxPrecip = headers.IndexOf("precipitation");

        if (idxTemp < 0) { error = $"Нет столбца temperature_2m"; return result; }

        for (int i = headerIndex + 1; i < lines.Length; i++)
        {
            if (string.IsNullOrWhiteSpace(lines[i])) continue;
            var fields = ParseCsvLine(lines[i]);
            if (fields.Count == 0 || !LooksLikeDate(fields[0])) continue;
            if (fields.Count <= idxTemp) continue;

            // Парсим время
            DateTime timestamp = DateTime.MinValue;
            if (idxTime >= 0 && idxTime < fields.Count)
                DateTime.TryParse(fields[idxTime], CultureInfo.InvariantCulture,
                    DateTimeStyles.RoundtripKind, out timestamp);

            float temp = ParseFloat(fields, idxTemp);
            float hum = ParseFloat(fields, idxHum);
            float rad = ParseFloat(fields, idxRad);
            float soil = ParseFloat(fields, idxSoil);
            float snow = ParseFloat(fields, idxSnow);
            float precip = idxPrecip >= 0 ? ParseFloat(fields, idxPrecip) : 0f;

            float solarEnergyMj = rad * 0.0036f;
            float soilMoisture = Math.Clamp((soil - 0.10f) / 0.28f, 0f, 1f);
            float effectiveSnowCm = (snow * 100f) * 0.5f;
            if (temp > 3f) effectiveSnowCm = 0f;
            float groundTemp = (temp < 0f && effectiveSnowCm > 0f)
                ? temp / (1f + 0.15f * effectiveSnowCm) : temp;
            float airHumidity = Math.Clamp(hum / 100f, 0f, 1f);

            result.Add(new WeatherData
            {
                Timestamp = timestamp,
                AirTemp = temp,
                GroundTemp = groundTemp,
                AirHumidity = airHumidity,
                SoilMoisture = soilMoisture,
                SolarEnergyMj = solarEnergyMj,
                EffectiveSnowCm = effectiveSnowCm,
                PrecipitationMm = precip
            });
        }

        if (result.Count == 0) error = "Ни одна строка данных не распознана";
        return result;
    }

    // "temperature_2m (°C)" → "temperature_2m"
    // '"hourly"'            → "hourly"
    private static string NormalizeHeader(string raw)
    {
        var s = raw.Trim().Trim('"');
        int paren = s.IndexOf('(');
        if (paren >= 0) s = s.Substring(0, paren);
        return s.Trim().ToLowerInvariant();
    }

    // "2000-01-01T00:00" → true;  "iso8601" → false
    private static bool LooksLikeDate(string first)
    {
        if (string.IsNullOrEmpty(first)) return false;
        char c0 = first[0];
        return c0 >= '0' && c0 <= '9';
    }

    private static List<string> ParseCsvLine(string line)
    {
        var result = new List<string>();
        bool inQuotes = false;
        var current = new StringBuilder();
        foreach (char c in line)
        {
            if (c == '"') inQuotes = !inQuotes;
            else if (c == ',' && !inQuotes) { result.Add(current.ToString()); current.Clear(); }
            else current.Append(c);
        }
        result.Add(current.ToString());
        return result;
    }

    private static float ParseFloat(List<string> fields, int index)
    {
        if (index < 0 || index >= fields.Count) return 0f;
        return float.TryParse(fields[index], NumberStyles.Float,
            CultureInfo.InvariantCulture, out var v) ? v : 0f;
    }
}.

/// <summary>
/// Ограничитель частоты кадров через дельта-время.
/// </summary>
public sealed class FpsLimiter
{
    private readonly double _targetFrameTimeSeconds;
    private readonly Stopwatch _stopwatch = new();
    private double _lastAccumulatedTime;

    public int TargetFps { get; }
    public float DeltaTime { get; private set; }
    public double ElapsedTime => _stopwatch.Elapsed.TotalSeconds;

    public FpsLimiter(int targetFps)
    {
        TargetFps = targetFps > 0 ? targetFps
            : throw new ArgumentOutOfRangeException(nameof(targetFps));
        _targetFrameTimeSeconds = 1.0 / targetFps;
    }

    /// <summary>Возвращает true, когда пора делать тик.</summary>
    public bool BeginTick()
    {
        if (!_stopwatch.IsRunning) _stopwatch.Start();

        double now = _stopwatch.Elapsed.TotalSeconds;
        double delta = now - _lastAccumulatedTime;

        if (delta < _targetFrameTimeSeconds * 0.95) return false;

        DeltaTime = (float)Math.Min(delta, _targetFrameTimeSeconds * 2);
        _lastAccumulatedTime = now;
        return true;
    }
}.

namespace AsciiFarm.Core;

/// <summary>
/// Блокиратор потока-владельца. AttachToCurrentThread() вызывается в начале потока,
/// после чего любой доступ из другого потока — исключение, а не гонка.
/// </summary>
public sealed class ThreadGuard
{
    private readonly string _name;
    private int _ownerThreadId;   // 0 = ещё не привязан
    private int _busy;            // 1 = данные прямо сейчас модифицируются

    public ThreadGuard(string name) => _name = name;

    public void AttachToCurrentThread() =>
        Interlocked.Exchange(ref _ownerThreadId, Environment.CurrentManagedThreadId);

    public void Validate()
    {
        int owner = Volatile.Read(ref _ownerThreadId);
        if (owner != 0 && owner != Environment.CurrentManagedThreadId)
            throw new InvalidOperationException(
                $"«{_name}»: данные принадлежат потоку {owner}, " +
                $"доступ из потока {Environment.CurrentManagedThreadId} запрещён.");
    }

    /// <summary>Критическая секция изменения данных (запрещает повторный вход).</summary>
    public Scope Enter()
    {
        Validate();
        if (Interlocked.Exchange(ref _busy, 1) != 0)
            throw new InvalidOperationException($"«{_name}»: данные ещё модифицируются.");
        return new Scope(this);
    }

    private void Release() => Volatile.Write(ref _busy, 0);

    public readonly struct Scope : IDisposable
    {
        private readonly ThreadGuard _guard;
        internal Scope(ThreadGuard guard) => _guard = guard;
        public void Dispose() => _guard.Release();
    }
    /// <summary>
    /// Сбрасывает владельца и флаг занятости.
    /// Вызывать после завершения потока-владельца, если данные больше не модифицируются.
    /// </summary>
    public void Detach()
    {
        Volatile.Write(ref _busy, 0);
        Volatile.Write(ref _ownerThreadId, 0);
    }
}.

    }

    {
        Guard.Validate();
        if (!_stores.TryGetValue(typeof(T), out var store))
        {
            store = new ComponentStore<T>();
            _stores[typeof(T)] = store;
        }
        return (ComponentStore<T>)store;
    }

    public void Add<T>(Entity entity, T component) where T : struct => Store<T>().Set(entity, component);

    // Запросы материализуются в массив: системы могут спокойно удалять компоненты в цикле.
    public Entity[] Query<TA>() where TA : struct => Store<TA>().Entities.ToArray();

    public Entity[] Query<TA, TB>() where TA : struct where TB : struct
    {
        IComponentStore a = Store<TA>(), b = Store<TB>();
        var (small, large) = a.Count <= b.Count ? (a, b) : (b, a);
        return small.Entities.Where(large.Has).ToArray();
    }

    {
        IComponentStore c = Store<TC>();
        return Query<TA, TB>().Where(c.Has).ToArray();
    }

using System.Collections.Generic;
using System.Diagnostics;
namespace AsciiFarm.Data;

public sealed class FpsCounter
{
    private const int SampleSize = 60;
    private readonly Queue<double> _frameTimes = new();
    private readonly object _gate = new();

    // Для реального FPS (рендер)
    private int _framesThisSecond = 0;
    private readonly Stopwatch _renderTimer = Stopwatch.StartNew();

    // Для TPS (симуляция)
    private int _ticksThisSecond = 0;
    private readonly Stopwatch _tickTimer = Stopwatch.StartNew();

    public float CurrentFps { get; private set; }
    public float AverageFps { get; private set; }
    public int RealFps { get; private set; }
    public int TicksPerSecond { get; private set; }

    public void RecordFrame(double timestampSeconds)
    {
        lock (_gate)
        {
            _frameTimes.Enqueue(timestampSeconds);
            if (_frameTimes.Count > SampleSize) _frameTimes.Dequeue();

            if (_frameTimes.Count >= 2)
            {
                var first = _frameTimes.Peek();
                var last = _frameTimes.ToArray()[^1];
                double duration = last - first;
                if (duration > 0.001)
                    CurrentFps = (float)((_frameTimes.Count - 1) / duration);
            }

            // Считаем реальный FPS (рендер)
            _framesThisSecond++;
            if (_renderTimer.ElapsedMilliseconds >= 1000)
            {
                RealFps = _framesThisSecond;
                _framesThisSecond = 0;
                _renderTimer.Restart();
            }
        }
    }

    public void RecordTick()
    {
        lock (_gate)
        {
            _ticksThisSecond++;
            if (_tickTimer.ElapsedMilliseconds >= 1000)
            {
                TicksPerSecond = _ticksThisSecond;
                _ticksThisSecond = 0;
                _tickTimer.Restart();
            }
        }
    }

    public void UpdateAverage(float instantFps)
    {
        lock (_gate)
        {
            AverageFps = AverageFps == 0 ? instantFps : AverageFps * 0.9f + instantFps * 0.1f;
        }
    }
}.

namespace AsciiFarm.Input;

public sealed class CommandChannel
{
    private readonly Queue<GameCommand> _queue = new();
    private readonly object _gate = new();
    private readonly int _capacity;
    private bool _closed;

    public CommandChannel(int capacity) => _capacity = capacity;

    public void Post(GameCommand command)
    {
        lock (_gate)
        {
            if (_closed || command == GameCommand.None) return;
            if (_queue.Count >= _capacity) _queue.Dequeue();
            _queue.Enqueue(command);
        }
    }

    public void DrainTo(List<GameCommand> target)
    {
        lock (_gate)
        {
            while (_queue.Count > 0) target.Add(_queue.Dequeue());
        }
    }

    public void Close()
    {
        lock (_gate) { _closed = true; _queue.Clear(); }
    }
}.

/// <summary>
/// Тело потока ввода. Ссылки на World нет по построению.
/// </summary>
public sealed class InputLoop
{
    private enum Modifier { None, Speed, FastForward }

    private readonly CommandChannel _channel;
    private readonly KeyBindings _bindings;
    private readonly int _pollDelayMs;
    private Modifier _lastModifier = Modifier.None;

    public InputLoop(CommandChannel channel, KeyBindings bindings, int pollDelayMs)
    {
        _channel = channel;
        _bindings = bindings;
        _pollDelayMs = pollDelayMs;
    }

    public void Run(CancellationToken token)
    {
        while (!token.IsCancellationRequested)
        {
            bool worked = false;
            while (Console.KeyAvailable)
            {
                var keyInfo = Console.ReadKey(intercept: true);
                var key = keyInfo.Key;

                // Модификаторы: запоминаем последний
                if (key == ConsoleKey.X) { _lastModifier = Modifier.Speed; worked = true; continue; }
                if (key == ConsoleKey.P) { _lastModifier = Modifier.FastForward; worked = true; continue; }

                // Цифры 1-4 с модификатором
                if (key >= ConsoleKey.D1 && key <= ConsoleKey.D4)
                {
                    int n = key - ConsoleKey.D1 + 1;
                    if (_lastModifier == Modifier.Speed)
                    {
                        var cmd = n switch
                        {
                            1 => GameCommand.Speed1,
                            2 => GameCommand.Speed2,
                            3 => GameCommand.Speed3,
                            4 => GameCommand.Speed4,
                            _ => GameCommand.None
                        };
                        _channel.Post(cmd);
                        _lastModifier = Modifier.None;
                        worked = true;
                        continue;
                    }
                    if (_lastModifier == Modifier.FastForward)
                    {
                        var cmd = n switch
                        {
                            1 => GameCommand.FastForward1,
                            2 => GameCommand.FastForward2,
                            3 => GameCommand.FastForward3,
                            4 => GameCommand.FastForward4,
                            _ => GameCommand.None
                        };
                        _channel.Post(cmd);
                        _lastModifier = Modifier.None;
                        worked = true;
                        continue;
                    }
                }

                // Сброс модификатора при любой другой клавише (кроме X/P)
                if (_lastModifier != Modifier.None && key != ConsoleKey.X && key != ConsoleKey.P)
                    _lastModifier = Modifier.None;

                // Обычные команды из биндингов
                if (_bindings.Map.TryGetValue(key, out var command))
                {
                    _channel.Post(command);
                    worked = true;
                }
            }
            if (!worked) Thread.Sleep(_pollDelayMs);
        }
    }
}.

﻿using System;
using System.Collections.Generic;
using System.Linq;
using System.Text;
using System.Threading.Tasks;
using System;
using System.ComponentModel;
using System.Runtime.InteropServices;
using Microsoft.Win32.SafeHandles;

namespace askiifarm.Rendering

    public sealed class DoubleBufferedConsole : IDisposable
    {
        private const int STD_OUTPUT_HANDLE = -11;

        private const uint GENERIC_READ = 0x80000000;
        private const uint GENERIC_WRITE = 0x40000000;
        private const uint FILE_SHARE_READ = 0x00000001;
        private const uint FILE_SHARE_WRITE = 0x00000002;
        private const uint CONSOLE_TEXTMODE_BUFFER = 1;

        // добавить поле
        private readonly AsciiFarm.Core.ThreadGuard _guard = new("Console");

        /// <summary>Вызывается один раз в потоке рендера.</summary>
        public void AttachToCurrentThread() => _guard.AttachToCurrentThread();

        [StructLayout(LayoutKind.Sequential)]
        private struct COORD
        {
            public short X;
            public short Y;
        }

        [StructLayout(LayoutKind.Sequential)]
        private struct SMALL_RECT
        {
            public short Left;
            public short Top;
            public short Right;
            public short Bottom;
        }

        [StructLayout(LayoutKind.Explicit, CharSet = CharSet.Unicode)]
        private struct CHAR_INFO
        {
            [FieldOffset(0)]
            public char UnicodeChar;

            [FieldOffset(0)]
            public byte AsciiChar;

            [FieldOffset(2)]
            public short Attributes;
        }


        [DllImport("kernel32.dll", SetLastError = true)]
        private static extern nint GetStdHandle(int nStdHandle);

        [DllImport("kernel32.dll", SetLastError = true)]
        private static extern COORD GetLargestConsoleWindowSize(SafeFileHandle hConsoleOutput);

        [DllImport("kernel32.dll", SetLastError = true)]
        [return: MarshalAs(UnmanagedType.Bool)]
        private static extern bool SetConsoleScreenBufferSize(
            SafeFileHandle hConsoleOutput,
            COORD dwSize);

        [DllImport("kernel32.dll", SetLastError = true)]
        [return: MarshalAs(UnmanagedType.Bool)]
        private static extern bool SetConsoleWindowInfo(
            SafeFileHandle hConsoleOutput,
            [MarshalAs(UnmanagedType.Bool)] bool bAbsolute,
            ref SMALL_RECT lpConsoleWindow);

        [DllImport("kernel32.dll", SetLastError = true, CharSet = CharSet.Unicode)]
        [return: MarshalAs(UnmanagedType.Bool)]
        private static extern bool WriteConsoleOutput(
            SafeFileHandle hConsoleOutput,
            CHAR_INFO[] lpBuffer,
            COORD dwBufferSize,
            COORD dwBufferCoord,
            ref SMALL_RECT lpConsoleWriteRegion);

        [DllImport("kernel32.dll", SetLastError = true)]
        private static extern SafeFileHandle CreateConsoleScreenBuffer(
            uint dwDesiredAccess,
            uint dwShareMode,
            nint lpSecurityAttributes,
            uint dwFlags,
            nint lpScreenBufferData);

        [DllImport("kernel32.dll", SetLastError = true)]
        [return: MarshalAs(UnmanagedType.Bool)]
        private static extern bool SetConsoleActiveScreenBuffer(
            SafeFileHandle hConsoleOutput);

        private readonly SafeFileHandle _stdout;

        private SafeFileHandle _front;
        private SafeFileHandle _back;

        private CHAR_INFO[] _cells;

        private int _width;
        private int _height;

        private bool _dirty;
        private bool _disposed;

        public int Width => _width;
        public int Height => _height;

        /// <summary>
        /// Цвет текста по умолчанию для Clear().
        /// </summary>
        public ConsoleColor DefaultForeground { get; set; }

        /// <summary>
        /// Цвет фона по умолчанию для Clear().
        /// </summary>
        public ConsoleColor DefaultBackground { get; set; }

        public DoubleBufferedConsole(
            int width,
            int height,
            ConsoleColor defaultForeground = ConsoleColor.Gray,
            ConsoleColor defaultBackground = ConsoleColor.Black)
        {
            if (!RuntimeInformation.IsOSPlatform(OSPlatform.Windows))
                throw new PlatformNotSupportedException("Поддерживается только Windows.");

            if (Console.IsOutputRedirected)
                throw new InvalidOperationException("Вывод консоли перенаправлен. Нужна настоящая консоль.");

            ValidateSize(width, height);

            _width = width;
            _height = height;

            DefaultForeground = defaultForeground;
            DefaultBackground = defaultBackground;

            _stdout = OpenStandardOutput();

            try
            {
                ValidateAgainstLargestWindowSize(width, height);

                _front = CreateBuffer(width, height);
                _back = CreateBuffer(width, height);

                _cells = new CHAR_INFO[width * height];

                FillCells(
                    _cells,
                    ' ',
                    MakeAttribute(DefaultForeground, DefaultBackground));

                // Показываем начальный пустой кадр.
                WriteBuffer(_front, _cells, width, height);
                SetActiveBuffer(_front);

                _dirty = false;

                try
                {
                    Console.CursorVisible = false;
                }
                catch
                {
                    // Не критично, если консоль не позволяет управлять курсором.
                }
            }
            catch
            {
                _front?.Dispose();
                _back?.Dispose();
                _stdout.Dispose();
                throw;
            }
        }

        /// <summary>
        /// Добавляет один символ в буфер.
        /// Если координаты вне диапазона, вызов игнорируется.
        /// </summary>
        public void PutChar(
            int x,
            int y,
            char character,
            ConsoleColor foreground,
            ConsoleColor background)
        {
            _guard.Validate();
            CheckNotDisposed();

            if ((uint)x >= (uint)_width || (uint)y >= (uint)_height)
                return;

            int index = y * _width + x;

            _cells[index].UnicodeChar = character;
            _cells[index].Attributes = MakeAttribute(foreground, background);

            _dirty = true;
        }

        /// <summary>
        /// Рисует текущий буфер.
        /// Если изменений не было, ничего не делает, если не задан force = true.
        /// </summary>
        public void Draw(bool force = false)
        {
            _guard.Validate();
            CheckNotDisposed();

            if (!_dirty && !force)
                return;

            // Пишем кадр в скрытый буфер и делаем его активным.
            WriteBuffer(_back, _cells, _width, _height);
            SetActiveBuffer(_back);

            SafeFileHandle temp = _front;
            _front = _back;
            _back = temp;

            _dirty = false;
        }

        /// <summary>
        /// Очищает буфер пробелами и цветами по умолчанию.
        /// </summary>
        public void Clear()
        {
            _guard.Validate();
            CheckNotDisposed();
            Fill(' ', DefaultForeground, DefaultBackground);
        }

        /// <summary>
        /// Заливает весь буфер одним символом и цветами.
        /// </summary>
        public void Fill(
            char character,
            ConsoleColor foreground,
            ConsoleColor background)
        {
            _guard.Validate();
            CheckNotDisposed();

            short attributes = MakeAttribute(foreground, background);
            FillCells(_cells, character, attributes);

            _dirty = true;
        }

        /// <summary>
        /// Заливает прямоугольник.
        /// Координаты — верхний левый угол.
        /// Прямоугольник автоматически обрезается по границам буфера.
        /// </summary>
        public void FillRect(
            int x,
            int y,
            int width,
            int height,
            char character,
            ConsoleColor foreground,
            ConsoleColor background)
        {
            _guard.Validate();
            CheckNotDisposed();

            if (width < 0 || height < 0)
                throw new ArgumentOutOfRangeException(nameof(width), "Размер не может быть отрицательным.");

            if (width == 0 || height == 0)
                return;

            int left = Math.Max(0, x);
            int top = Math.Max(0, y);

            int right = (int)Math.Min(_width, (long)x + width);
            int bottom = (int)Math.Min(_height, (long)y + height);

            if (left >= right || top >= bottom)
                return;

            short attributes = MakeAttribute(foreground, background);

            for (int yy = top; yy < bottom; yy++)
            {
                int rowStart = yy * _width;

                for (int xx = left; xx < right; xx++)
                {
                    int index = rowStart + xx;

                    _cells[index].UnicodeChar = character;
                    _cells[index].Attributes = attributes;
                }
            }

            _dirty = true;
        }

        /// <summary>
        /// Меняет размер буфера и окна консоли.
        /// Текущий кадр при этом сбрасывается.
        /// </summary>
        public void Resize(int width, int height)
        {
            _guard.Validate();
            CheckNotDisposed();
            ValidateSize(width, height);

            if (width == _width && height == _height)
                return;

            ValidateAgainstLargestWindowSize(width, height);

            SafeFileHandle newFront = CreateBuffer(width, height);
            SafeFileHandle newBack = null;

            try
            {
                newBack = CreateBuffer(width, height);

                CHAR_INFO[] newCells = new CHAR_INFO[width * height];

                FillCells(
                    newCells,
                    ' ',
                    MakeAttribute(DefaultForeground, DefaultBackground));

                WriteBuffer(newFront, newCells, width, height);
                SetActiveBuffer(newFront);

                SafeFileHandle oldFront = _front;
                SafeFileHandle oldBack = _back;

                _front = newFront;
                _back = newBack;
                _cells = newCells;

                _width = width;
                _height = height;

                _dirty = false;

                oldFront?.Dispose();
                oldBack?.Dispose();
            }
            catch
            {
                newFront.Dispose();
                newBack?.Dispose();
                throw;
            }
        }

        public void Dispose()
        {
            if (_disposed)
                return;

            _disposed = true;

            try
            {
                // Возвращаем исходный консольный буфер.
                if (_stdout != null && !_stdout.IsInvalid)
                    SetConsoleActiveScreenBuffer(_stdout);
            }
            catch
            {
                // Если не удалось вернуть исходный буфер, игнорируем.
            }

            _front?.Dispose();
            _back?.Dispose();
            _stdout?.Dispose();

            try
            {
                Console.CursorVisible = true;
            }
            catch
            {
                // Не критично.
            }
        }

        private void CheckNotDisposed()
        {
            if (_disposed)
                throw new ObjectDisposedException(nameof(DoubleBufferedConsole));
        }

        private static void ValidateSize(int width, int height)
        {
            if (width <= 0 || height <= 0)
                throw new ArgumentOutOfRangeException(nameof(width), "Размеры должны быть положительными.");

            if (width > short.MaxValue || height > short.MaxValue)
                throw new ArgumentOutOfRangeException(nameof(width), "Размеры слишком большие для COORD.");

            if ((long)width * height > int.MaxValue)
                throw new ArgumentOutOfRangeException(nameof(width), "Слишком много ячеек.");
        }

        private void ValidateAgainstLargestWindowSize(int width, int height)
        {
            COORD largest = GetLargestConsoleWindowSize(_stdout);

            if (largest.X > 0 && largest.Y > 0 && (width > largest.X || height > largest.Y))
            {
                throw new ArgumentOutOfRangeException(
                    nameof(width),
                    $"Размер окна консоли не может быть больше {largest.X}x{largest.Y}.");
            }
        }

        private static SafeFileHandle OpenStandardOutput()
        {
            SafeFileHandle handle = new SafeFileHandle(GetStdHandle(STD_OUTPUT_HANDLE), false);

            if (handle.IsInvalid)
                throw new Win32Exception(Marshal.GetLastWin32Error());

            return handle;
        }

        private static SafeFileHandle CreateBuffer(int width, int height)
        {
            SafeFileHandle handle = CreateConsoleScreenBuffer(
                GENERIC_READ | GENERIC_WRITE,
                FILE_SHARE_READ | FILE_SHARE_WRITE,
                nint.Zero,
                CONSOLE_TEXTMODE_BUFFER,
                nint.Zero);

            if (handle.IsInvalid)
                throw new Win32Exception(Marshal.GetLastWin32Error());

            try
            {
                COORD size = new COORD
                {
                    X = (short)width,
                    Y = (short)height
                };

                SMALL_RECT target = new SMALL_RECT
                {
                    Left = 0,
                    Top = 0,
                    Right = (short)(width - 1),
                    Bottom = (short)(height - 1)
                };

                // Если окно можно сразу сделать нужного размера — делаем.
                // Если нет, сначала уменьшаем до 1x1, затем меняем размер буфера.
                if (!SetConsoleWindowInfo(handle, true, ref target))
                {
                    SMALL_RECT tiny = new SMALL_RECT
                    {
                        Left = 0,
                        Top = 0,
                        Right = 0,
                        Bottom = 0
                    };

                    if (!SetConsoleWindowInfo(handle, true, ref tiny))
                        throw new Win32Exception(Marshal.GetLastWin32Error());

                    if (!SetConsoleScreenBufferSize(handle, size))
                        throw new Win32Exception(Marshal.GetLastWin32Error());

                    if (!SetConsoleWindowInfo(handle, true, ref target))
                        throw new Win32Exception(Marshal.GetLastWin32Error());
                }
                else
                {
                    if (!SetConsoleScreenBufferSize(handle, size))
                        throw new Win32Exception(Marshal.GetLastWin32Error());
                }

                return handle;
            }
            catch
            {
                handle.Dispose();
                throw;
            }
        }

        private static void WriteBuffer(
            SafeFileHandle handle,
            CHAR_INFO[] cells,
            int width,
            int height)
        {
            COORD bufferSize = new COORD
            {
                X = (short)width,
                Y = (short)height
            };

            COORD bufferCoord = new COORD
            {
                X = 0,
                Y = 0
            };

            SMALL_RECT region = new SMALL_RECT
            {
                Left = 0,
                Top = 0,
                Right = (short)(width - 1),
                Bottom = (short)(height - 1)
            };

            if (!WriteConsoleOutput(handle, cells, bufferSize, bufferCoord, ref region))
                throw new Win32Exception(Marshal.GetLastWin32Error());
        }

        private static void SetActiveBuffer(SafeFileHandle handle)
        {
            if (!SetConsoleActiveScreenBuffer(handle))
                throw new Win32Exception(Marshal.GetLastWin32Error());
        }

        private static void FillCells(
            CHAR_INFO[] cells,
            char character,
            short attributes)
        {
            for (int i = 0; i < cells.Length; i++)
            {
                cells[i].UnicodeChar = character;
                cells[i].Attributes = attributes;
            }
        }

        private static short MakeAttribute(ConsoleColor foreground, ConsoleColor background)
        {
            int fg = (int)foreground;
            int bg = (int)background;

            if ((fg & ~0xF) != 0 || (bg & ~0xF) != 0)
            {
                throw new ArgumentOutOfRangeException(
                    nameof(foreground),
                    "ConsoleColor должен быть в диапазоне 0..15.");
            }

            return (short)(fg | bg << 4);
        }
    }

using AsciiFarm.Core;
namespace AsciiFarm.Rendering;

/// <summary>
/// Тройная буферизация: write-буфер принадлежит симуляции, read-буфер — рендеру,
/// средний слот меняется только под lock. Одновременного доступа к содержимому нет физически.
/// </summary>
public sealed class FrameExchange<T> where T : class
{
    private readonly T[] _buffers;
    private readonly object _gate = new();
    private readonly ThreadGuard _writerGuard = new("FrameExchange.Writer");
    private readonly ThreadGuard _readerGuard = new("FrameExchange.Reader");

    private int _writeIndex;        // трогает только писатель
    private int _readIndex = 1;     // трогает только читатель
    private int _readyIndex = 2;    // только под _gate
    private bool _hasFreshFrame;    // флаг готовности, только под _gate

    public FrameExchange(Func<T> factory) => _buffers = new[] { factory(), factory(), factory() };

    public void AttachWriter() => _writerGuard.AttachToCurrentThread();
    public void AttachReader() => _readerGuard.AttachToCurrentThread();

    public T WriteBuffer
    {
        get { _writerGuard.Validate(); return _buffers[_writeIndex]; }
    }

    public void Publish()
    {
        _writerGuard.Validate();
        lock (_gate)
        {
            (_writeIndex, _readyIndex) = (_readyIndex, _writeIndex);
            _hasFreshFrame = true;
        }
    }

    public bool TryAcquireFresh(out T frame)
    {
        _readerGuard.Validate();
        lock (_gate)
        {
            if (_hasFreshFrame)
            {
                (_readIndex, _readyIndex) = (_readyIndex, _readIndex);
                _hasFreshFrame = false;
                frame = _buffers[_readIndex];
                return true;
            }
        }
        frame = _buffers[_readIndex];
        return false;
    }
}.

﻿using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using System.Reflection;
using System.Text;

namespace AsciiFarm.Serialization;

/// <summary>
/// Компонентная система сохранения. Формат файла:
/// [6 байт: сигнатура "AFARM2"]
/// [4 байта: версия формата]
/// [4 байта: количество записей]
/// Для каждой записи:
///   [строка: имя типа компонента]
///   [4 байта: версия компонента]
///   [4 байта: ID сущности]
///   [4 байта: длина данных]
///   [N байт: данные]
/// 
/// Forward compatibility: неизвестные компоненты пропускаются.
/// Backward compatibility: старые компоненты читаются через миграции.
/// </summary>
public sealed class ComponentSaveSystem
{
    private static readonly byte[] Signature = Encoding.ASCII.GetBytes("AFARM2");
    public const int FormatVersion = 1;

    private readonly ComponentSerializerRegistry _registry;

    public ComponentSaveSystem(ComponentSerializerRegistry registry) => _registry = registry;

    public void Save(string filePath, World world)
    {
        using var stream = new FileStream(filePath, FileMode.Create, FileAccess.Write);
        using var writer = new BinaryWriter(stream, Encoding.UTF8);

        writer.Write(Signature);
        writer.Write(FormatVersion);

        var entries = new List<(string typeName, int version, int entityId, byte[] data)>();

        foreach (var (componentType, store) in world.AllStores())
        {
            // Пропускаем компоненты с [DontSave]
            if (componentType.GetCustomAttribute<DontSaveAttribute>() != null)
                continue;

            var serializer = _registry.Get(componentType);
            if (serializer == null) continue; // нет сериализатора — не сохраняем

            foreach (var (entity, component) in store.AllComponentsBoxed())
            {
                using var ms = new MemoryStream();
                using var compWriter = new BinaryWriter(ms, Encoding.UTF8, leaveOpen: true);
                serializer.Serialize(compWriter, component);
                entries.Add((serializer.TypeName, serializer.Version, entity.Id, ms.ToArray()));
            }
        }

        writer.Write(entries.Count);
        foreach (var (typeName, version, entityId, data) in entries)
        {
            writer.Write(typeName);
            writer.Write(version);
            writer.Write(entityId);
            writer.Write(data.Length);
            writer.Write(data);
        }
    }

    public void Load(string filePath, World world)
    {
        using var stream = new FileStream(filePath, FileMode.Open, FileAccess.Read);
        using var reader = new BinaryReader(stream, Encoding.UTF8);

        var sig = reader.ReadBytes(6);
        if (!sig.AsSpan().SequenceEqual(Signature))
            throw new InvalidOperationException($"Неверная сигнатура файла сохранения");

        int formatVersion = reader.ReadInt32();
        if (formatVersion > FormatVersion)
            throw new InvalidOperationException($"Формат {formatVersion} новее поддерживаемого {FormatVersion}");

        int count = reader.ReadInt32();
        int maxEntityId = 0;

        for (int i = 0; i < count; i++)
        {
            string typeName = reader.ReadString();
            int version = reader.ReadInt32();
            int entityId = reader.ReadInt32();
            int dataLen = reader.ReadInt32();
            byte[] data = reader.ReadBytes(dataLen);

            var serializer = _registry.Get(typeName);
            if (serializer == null)
            {
                // Неизвестный компонент — пропускаем (forward compatibility!)
                continue;
            }

            if (version > serializer.Version)
            {
                // Версия компонента новее — пропускаем (forward compatibility!)
                continue;
            }

            using var ms = new MemoryStream(data);
            using var compReader = new BinaryReader(ms, Encoding.UTF8, leaveOpen: true);
            var component = serializer.Deserialize(compReader, version);

            var entity = new Entity(entityId);

            // Добавляем компонент через рефлексию
            var addMethod = typeof(World).GetMethod(nameof(World.Add))!.MakeGenericMethod(serializer.ComponentType);
            addMethod.Invoke(world, new[] { entity, component });

            if (entityId > maxEntityId) maxEntityId = entityId;
        }

        // Резервируем ID, чтобы новые сущности не конфликтовали
        world.ReserveId(maxEntityId);
    }

    public static bool IsNewFormat(string filePath)
    {
        if (!File.Exists(filePath)) return false;
        using var stream = File.OpenRead(filePath);
        if (stream.Length < 6) return false;
        var sig = new byte[6];
        stream.Read(sig, 0, 6);
        return sig.AsSpan().SequenceEqual(Signature);
    }

﻿using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using System.Text;
namespace askiifarm.Serialization
{
    public class GameStorage
    {
        private static readonly byte[] BinarySignature = Encoding.ASCII.GetBytes("AFARM");
        private readonly Dictionary<Version, IVersionFormatHandler> _handlers = new();
        private readonly Dictionary<Version, IVersionConverter> _converters = new();

        public static Version LatestVersion => new Version(0, 0, 3);

        public GameStorage()
        {
            RegisterHandler(new FormatHandlerV0_0_0());
            RegisterHandler(new FormatHandlerV0_0_1());
            RegisterHandler(new FormatHandlerV0_0_2());
            RegisterHandler(new FormatHandlerV0_0_3());

            RegisterConverter(new ConverterV0_0_0ToV0_0_1());
            RegisterConverter(new ConverterV0_0_1ToV0_0_2());
            RegisterConverter(new ConverterV0_0_2ToV0_0_3());
            RegisterConverter(new ConverterV0_0_3ToV0_0_2());
        }

        public void RegisterHandler(IVersionFormatHandler handler) =>
            _handlers[handler.Version] = handler;

        public void RegisterConverter(IVersionConverter converter) =>
            _converters[converter.SourceVersion] = converter;

        public void Save(string filePath, Version version, object data)
        {
            if (!_handlers.TryGetValue(version, out var handler))
                throw new NotSupportedException($"Обработчик для версии {version} не найден.");
            using var stream = new FileStream(filePath, FileMode.Create, FileAccess.Write);
            if (version > new Version(0, 0, 0))
            {
                using var headerWriter = new BinaryWriter(stream, Encoding.UTF8, leaveOpen: true);
                headerWriter.Write(BinarySignature);
                headerWriter.Write(version.Major);
                headerWriter.Write(version.Minor);
                headerWriter.Write(version.Build);
            }
            handler.Save(stream, data);
        }

        public (Version Version, object Data) Load(string filePath)
        {
            using var stream = new FileStream(filePath, FileMode.Open, FileAccess.Read);
            Version detectedVersion;
            if (HasBinarySignature(stream))
            {
                using var headerReader = new BinaryReader(stream, Encoding.UTF8, leaveOpen: true);
                int major = headerReader.ReadInt32();
                int minor = headerReader.ReadInt32();
                int build = headerReader.ReadInt32();
                detectedVersion = new Version(major, minor, build);
            }
            else
            {
                stream.Position = 0;
                detectedVersion = new Version(0, 0, 0);
            }
            if (!_handlers.TryGetValue(detectedVersion, out var handler))
                throw new NotSupportedException($"Неизвестная версия сохранения: {detectedVersion}");
            object data = handler.Load(stream);
            return (detectedVersion, data);
        }

        public object ConvertStep(Version currentVersion, object data)
        {
            if (!_converters.TryGetValue(currentVersion, out var converter))
                throw new InvalidOperationException($"Нет конвертера для перехода с версии {currentVersion}.");
            return converter.Convert(data);
        }

        public T LoadAndConvert<T>(string filePath, Version targetVersion) where T : class
        {
            var (currentVersion, currentData) = Load(filePath);
            while (currentVersion < targetVersion)
            {
                if (!_converters.TryGetValue(currentVersion, out var converter))
                    throw new InvalidOperationException($"Не найден конвертер для версии {currentVersion}.");
                currentData = converter.Convert(currentData);
                currentVersion = converter.TargetVersion;
            }
            if (currentVersion != targetVersion)
                throw new InvalidOperationException($"Версия файла ({currentVersion}) выше целевой ({targetVersion}).");
            return currentData as T
                ?? throw new InvalidCastException($"Данные версии {targetVersion} не соответствуют типу {typeof(T).Name}.");
        }

        public T LoadLatest<T>(string filePath) where T : class
        {
            return LoadAndConvert<T>(filePath, LatestVersion);
        }

        /// <summary>
        /// Сохраняет данные. Если версия файла совпадает — перезаписывает.
        /// Если файл старее — конвертирует прочитанное и перезаписывает тем же путём.
        /// Если файл новее — создаёт новый файл с суффиксом версии.
        /// </summary>
        public void SaveSmart(string filePath, Version version, object data)
        {
            if (!File.Exists(filePath))
            {
                Save(filePath, version, data);
                return;
            }

            var (existingVersion, _) = Load(filePath);

            if (existingVersion == version)
            {
                // Та же версия — просто перезаписываем
                Save(filePath, version, data);
            }
            else if (existingVersion < version)
            {
                // Файл старее: перезаписываем его новой версией
                Save(filePath, version, data);
            }
            else
            {
                // Файл новее — создаём отдельный файл с суффиксом
                string dir = Path.GetDirectoryName(filePath) ?? ".";
                string name = Path.GetFileNameWithoutExtension(filePath);
                string ext = Path.GetExtension(filePath);
                string newFilePath = Path.Combine(dir, $"{name}_v{version.Major}.{version.Minor}.{version.Build}{ext}");
                Save(newFilePath, version, data);
            }
        }

        private static bool HasBinarySignature(Stream stream)
        {
            if (stream.Length < BinarySignature.Length)
                return false;
            byte[] buffer = new byte[BinarySignature.Length];
            int bytesRead = stream.Read(buffer, 0, buffer.Length);
            return bytesRead == BinarySignature.Length && buffer.SequenceEqual(BinarySignature);
        }
    }

            for (int y = 0; y < height; y++)

        {
            using var reader = new StreamReader(stream, Encoding.UTF8, leaveOpen: true);
            string content = reader.ReadToEnd();
            string[] lines = content.Split(new[] { "\r\n", "\n", "\r" }, StringSplitOptions.None);
            int height = lines.Length;
            while (height > 0 && string.IsNullOrEmpty(lines[height - 1])) height--;
            if (height == 0) return new GameDataV0_0_0 { Map = new char[0, 0] };
            int width = 0;
            for (int y = 0; y < height; y++)
                if (lines[y].Length > width) width = lines[y].Length;
            char[,] map = new char[width, height];
            for (int y = 0; y < height; y++)
            {
                string line = lines[y];
                for (int x = 0; x < width; x++)
                    map[x, y] = x < line.Length ? line[x] : ' ';
            }
            return new GameDataV0_0_0 { Map = map };
        }

            }

        _fpsCounter.RecordFrame(frame.Tick);  // ДОБАВИТЬ
    }

using AsciiFarm.Core;
using AsciiFarm.Data;
using System.Diagnostics;
namespace AsciiFarm.Systems;

[GameSystem(Order = 1)]
public sealed class ElapsedTimeSystem : ISystem
{
    private readonly Stopwatch _tickTimer = Stopwatch.StartNew();
    private double _lastTickTime = 0;

    public void Update(World world)
    {
        var state = world.Resource<GameState>();
        var fpsCounter = world.Resource<FpsCounter>();

        double now = _tickTimer.Elapsed.TotalSeconds;
        double realDeltaTime = now - _lastTickTime;
        _lastTickTime = now;
        realDeltaTime = Math.Min(realDeltaTime, 0.1);

        // Игровое время с учётом скорости
        double effectiveDeltaTime = realDeltaTime * state.TimeSpeed;
        effectiveDeltaTime = Math.Min(effectiveDeltaTime, state.MaxTimeStep);

        state.DeltaTime = (float)realDeltaTime;
        state.ElapsedTime += realDeltaTime;
        state.GameTimeElapsed += effectiveDeltaTime;
        state.Tick++;

        fpsCounter.RecordTick();
    }
}.

using AsciiFarm.Core;
using AsciiFarm.Data;
namespace AsciiFarm.Systems;

/// <summary>Бывший MapPhysics: проходимость берётся из данных палитры, а не из switch.</summary>
[GameSystem(Order = 4)]
public sealed class MovementSystem : ISystem
{
    public void Update(World world)
    {
        var maps = world.Query<TileMapComponent>();
        if (maps.Length == 0) return;

        var map = world.Store<TileMapComponent>().Get(maps[0]);
        var palette = world.Resource<TilePalette>();
        var positions = world.Store<Position>();
        var intents = world.Store<MoveIntent>();

        foreach (var entity in world.Query<Position, MoveIntent>())
        {
            var intent = intents.Get(entity);
            var position = positions.Get(entity);
            int targetX = position.X + intent.Dx;
            int targetY = position.Y + intent.Dy;

            if (IsWalkable(map, palette, targetX, targetY))
                positions.Set(entity, new Position(targetX, targetY));

            intents.Remove(entity);   // намерение одноразовое
        }
    }

    private static bool IsWalkable(TileMapComponent map, TilePalette palette, int x, int y)
    {
        if (x < 0 || y < 0 || x >= map.Width || y >= map.Height) return false;
        return !palette.Resolve(map.Tiles[x, y]).BlocksMovement;
    }
}.

using AsciiFarm.Components;
using AsciiFarm.Core;
using AsciiFarm.Data;
using AsciiFarm.Input;
namespace AsciiFarm.Systems;

[GameSystem(Order = 3)]
public sealed class PlayerIntentSystem : ISystem
{
    public void Update(World world)
    {
        var state = world.Resource<GameState>();
        var commands = world.Resource<CommandBuffer>().Items;
        var intents = world.Store<MoveIntent>();
        var players = world.Query<PlayerTag, Position>();

        // Сбрасываем все предыдущие намерения
        foreach (var entity in intents.Entities.ToArray())
            intents.Remove(entity);

        // Накапливаем вектор движения из всех команд за тик
        int totalDx = 0;
        int totalDy = 0;
        bool exitRequested = false;

        foreach (var command in commands)
        {
            if (command == GameCommand.Exit)
            {
                exitRequested = true;
                continue;
            }

            var (dx, dy) = command switch
            {
                GameCommand.MoveUp => (0, -1),
                GameCommand.MoveDown => (0, 1),
                GameCommand.MoveLeft => (-1, 0),
                GameCommand.MoveRight => (1, 0),
                _ => (0, 0)
            };
            totalDx += dx;
            totalDy += dy;
        }

        // Ограничиваем каждый компонент значением [-1, 1]
        totalDx = Math.Clamp(totalDx, -1, 1);
        totalDy = Math.Clamp(totalDy, -1, 1);

        // Если есть движение, устанавливаем намерение
        if (totalDx != 0 || totalDy != 0)
        {
            foreach (var entity in players)
                intents.Set(entity, new MoveIntent(totalDx, totalDy));
        }

        if (exitRequested)
        {
            state.IsRunning = false;
            state.NeedsSave = true;
        }

        // Обработка команд управления временем
        foreach (var command in commands)
        {
            switch (command)
            {
                case GameCommand.Speed1:
                    state.TimeSpeed = 1.0f;
                    break;
                case GameCommand.Speed2:
                    state.TimeSpeed = 30.0f; // 48 реальных минут = 1 игровой день (24 часа)
                    break;
                case GameCommand.Speed3:
                    state.TimeSpeed = 1440.0f; // 1 реальная минута = 1 игровой день
                    break;
                case GameCommand.Speed4:
                    state.TimeSpeed = 43200.0f; // 1 реальная минута = 1 игровой месяц (30 дней)
                    break;
                case GameCommand.FastForward1:
                    state.FastForwardUntil = state.ElapsedTime + 3600.0; // +1 час реального времени
                    break;
                case GameCommand.FastForward2:
                    state.FastForwardUntil = state.ElapsedTime + 86400.0; // +1 день
                    break;
                case GameCommand.FastForward3:
                    state.FastForwardUntil = state.ElapsedTime + 604800.0; // +1 неделя
                    break;
                case GameCommand.FastForward4:
                    state.FastForwardUntil = state.ElapsedTime + 2592000.0; // +1 месяц (30 дней)
                    break;
            }
        }

        commands.Clear();
    }
}.

using System.Collections.Generic;
using AsciiFarm.Components;
using AsciiFarm.Core;
using AsciiFarm.Data;

/// <summary>
/// Анимация реки: каждый водный тайл с шансом 80% синий, 20% белый.
/// Псевдослучайность детерминирована по (x, y, time).
/// </summary>
[GameSystem(Order = 5)]
public sealed class RiverFlowSystem : ISystem
{
    private const char WaterSymbol = '≈';
    private const float AnimationSpeed = 1f;
    private const float BlueProbability = 0.5f;

    public void Update(World world)
    {
        var state = world.Resource<GameState>();
        var maps = world.Query<TileMapComponent>();
        if (maps.Length == 0) return;

        var map = world.Store<TileMapComponent>().Get(maps[0]);
        float time = (float)state.ElapsedTime;

        var waterColors = new Dictionary<(int, int), TileStyle>();

        for (int y = 0; y < map.Height; y++)
        {
            for (int x = 0; x < map.Width; x++)
            {
                if (map.Tiles[x, y] != WaterSymbol) continue;

                // Детерминированный псевдослучайный выбор цвета
                int hash = HashCode.Combine(x, y, (int)(time * AnimationSpeed));
                float randomValue = (hash & 0xFFFF) / (float)0xFFFF;

                ConsoleColor fg = randomValue < BlueProbability
                    ? ConsoleColor.Cyan
                    : ConsoleColor.White;

                waterColors[(x, y)] = new TileStyle(fg, ConsoleColor.Black, false);
            }
        }

        world.SetResource(new WaterColorCache(waterColors));
    }
}

public sealed class WaterColorCache
{

    public WaterColorCache(Dictionary<(int, int), TileStyle> colors) => Colors = colors;

using AsciiFarm.Components;
using AsciiFarm.Core;
using AsciiFarm.Data;
using AsciiFarm.Rendering;
namespace AsciiFarm.Systems;

public sealed class SnapshotSystem
{
    private readonly FrameExchange<RenderFrame> _frames;

    public SnapshotSystem(FrameExchange<RenderFrame> frames) => _frames = frames;

    public void Update(World world)
    {
        var frame = _frames.WriteBuffer;
        var palette = world.Resource<TilePalette>();
        var gameState = world.Resource<GameState>();
        var characterPalette = world.Resource<CharacterPalette>();
        var fpsCounter = world.Resource<FpsCounter>();

        WaterColorCache? waterCache = null;
        try { waterCache = world.Resource<WaterColorCache>(); } catch { /* первый кадр */ }

        Array.Fill(frame.Symbols, ' ');
        Array.Fill(frame.Foreground, palette.Fallback.Foreground);
        Array.Fill(frame.Background, palette.Fallback.Background);

        foreach (var entity in world.Query<TileMapComponent>())
            DrawTiles(frame, world.Store<TileMapComponent>().Get(entity), palette, waterCache);

        var glyphs = world.Store<Glyph>();
        var positions = world.Store<Position>();
        var playerTags = world.Store<PlayerTag>();

        foreach (var entity in world.Query<Glyph, Position>().OrderBy(e => glyphs.Get(e).Layer))
        {
            var glyph = glyphs.Get(entity);
            var pos = positions.Get(entity);

            if (playerTags.Has(entity))
                glyph = characterPalette.Get(characterPalette.DefaultPlayerKey);

            Put(frame, pos.X, pos.Y, glyph.Symbol, glyph.Foreground, glyph.Background);
        }

        DrawUI(frame, gameState, fpsCounter);
        DrawWeatherUI(frame, world.Resource<WeatherCurrent>(), world.Resource<WeatherDataSource>());
        DrawGameTime(frame, gameState);
        frame.Tick = gameState.Tick;
        _frames.Publish();
    }

    private static void DrawTiles(RenderFrame frame, TileMapComponent map,
                                  TilePalette palette, WaterColorCache? waterCache)
    {
        int width = Math.Min(map.Width, frame.Width);
        int height = Math.Min(map.Height, frame.Height);
        for (int y = 0; y < height; y++)
            for (int x = 0; x < width; x++)
            {
                char tile = map.Tiles[x, y];
                var style = waterCache != null && waterCache.Colors.TryGetValue((x, y), out var ws)
                    ? ws
                    : palette.Resolve(tile);

                Put(frame, x, y, tile, style.Foreground, style.Background);
            }
    }

    // В методе DrawUI:
    private static void DrawUI(RenderFrame frame, GameState state, FpsCounter fpsCounter)
    {
        string fpsStr = $"FPS: {fpsCounter.RealFps}";
        string tickStr = $"TPS: {fpsCounter.TicksPerSecond}";

        int startX = frame.Width - Math.Max(fpsStr.Length, tickStr.Length) - 1;

        for (int i = 0; i < fpsStr.Length; i++)
            if (startX + i < frame.Width)
                Put(frame, startX + i, 0, fpsStr[i], ConsoleColor.Green, ConsoleColor.Black);

        for (int i = 0; i < tickStr.Length; i++)
            if (startX + i < frame.Width && 1 < frame.Height)
                Put(frame, startX + i, 1, tickStr[i], ConsoleColor.Cyan, ConsoleColor.Black);
    }
    // Новый метод:
    private static void DrawWeatherUI(RenderFrame frame, WeatherCurrent weather, WeatherDataSource source)
    {
        int panelWidth = 24;
        int startX = frame.Width - panelWidth - 2;
        int startY = 3;

        (string text, ConsoleColor color)[] lines;

        if (!weather.HasData)
        {
            var err = source.LoadError ?? "Данные не загружены";
            if (err.Length > 22) err = err.Substring(0, 22);
            lines = new (string, ConsoleColor)[]
            {
            ("── ПОГОДА ──", ConsoleColor.DarkGray),
            ("  нет данных", ConsoleColor.DarkGray),
            (err, ConsoleColor.Red),
            };
        }
        else
        {
            var d = weather.Current;
            lines = new (string, ConsoleColor)[]
            {
            ("── ПОГОДА ──", ConsoleColor.Cyan),
            ($"Час: {weather.HourIndex:D2}", ConsoleColor.DarkGray),
            ($"Воздух:{d.AirTemp,7:F1}°C", ConsoleColor.White),
            ($"Земля: {d.GroundTemp,7:F1}°C", d.GroundTemp < 0 ? ConsoleColor.Blue : ConsoleColor.White),
            ($"Влажн: {d.AirHumidity * 100,6:F0}%", ConsoleColor.White),
            ($"Почва: {d.SoilMoisture * 100,6:F0}%", d.SoilMoisture < 0.2f ? ConsoleColor.Red : ConsoleColor.Green),
            ($"Солнц: {d.SolarEnergyMj,6:F2}МДж", ConsoleColor.Yellow),
            ($"Снег:  {d.EffectiveSnowCm,6:F1}см", ConsoleColor.White),
            ($"Осад:  {d.PrecipitationMm,6:F1}мм", ConsoleColor.Blue),
            };
        }

        for (int i = 0; i < lines.Length; i++)
        {
            var (text, color) = lines[i];
            for (int j = 0; j < text.Length && startX + j < frame.Width; j++)
            {
                if (startY + i < frame.Height)
                    Put(frame, startX + j, startY + i, text[j], color, ConsoleColor.Black);
            }
        }
    }

    // В методе DrawGameTime:
    private static void DrawGameTime(RenderFrame frame, GameState state)
    {
        // Игровое время из GameTimeElapsed
        DateTime gameTime = state.GameStartDate.AddSeconds(state.GameTimeElapsed);
        string dateStr = gameTime.ToString("dd.MM.yyyy");
        string timeStr = gameTime.ToString("HH:mm:ss");
        string speedStr = $"x{state.TimeSpeed:F0}";

        // Реальное время, прошедшее с запуска
        TimeSpan realTime = TimeSpan.FromSeconds(state.ElapsedTime);
        string realTimeStr = $"{(int)realTime.TotalHours:D2}:{realTime.Minutes:D2}:{realTime.Seconds:D2}";

        int startX = frame.Width - 12;  // Сдвинуто вправо
        int startY = frame.Height - 3;

        for (int i = 0; i < dateStr.Length && startX + i < frame.Width; i++)
            Put(frame, startX + i, startY, dateStr[i], ConsoleColor.White, ConsoleColor.Black);
        for (int i = 0; i < timeStr.Length && startX + i < frame.Width; i++)
            Put(frame, startX + i, startY + 1, timeStr[i], ConsoleColor.Yellow, ConsoleColor.Black);
        for (int i = 0; i < speedStr.Length && startX + i < frame.Width; i++)
            Put(frame, startX + i, startY + 2, speedStr[i], ConsoleColor.Green, ConsoleColor.Black);
        for (int i = 0; i < realTimeStr.Length && startX + i < frame.Width; i++)
            Put(frame, startX + i, startY + 3, realTimeStr[i], ConsoleColor.DarkGray, ConsoleColor.Black);
    }
    private static void Put(RenderFrame frame, int x, int y, char symbol, ConsoleColor fg, ConsoleColor bg)
    {
        if ((uint)x >= (uint)frame.Width || (uint)y >= (uint)frame.Height) return;
        int index = y * frame.Width + x;
        frame.Symbols[index] = symbol;
        frame.Foreground[index] = fg;
        frame.Background[index] = bg;
    }
}.

using System;
using System.Linq;
using System.Reflection;
using AsciiFarm.Core;
namespace AsciiFarm.Systems;

public static class SystemPipeline
{
    public static ISystem[] CreateSimulationSystems()
    {
        return Assembly.GetExecutingAssembly()
            .GetTypes()
            .Where(t => typeof(ISystem).IsAssignableFrom(t) && !t.IsAbstract && !t.IsInterface)
            .Select(t => new { Type = t, Attr = t.GetCustomAttribute<GameSystemAttribute>() })
            .Where(x => x.Attr != null)
            .OrderBy(x => x.Attr.Order)
            .Select(x => (ISystem)(Activator.CreateInstance(x.Type) ?? throw new InvalidOperationException($"Не удалось создать экземпляр {x.Type.Name}")))
            .ToArray();
    }
}.

using System;
using AsciiFarm.Core;
using AsciiFarm.Data;
namespace AsciiFarm.Systems;

[GameSystem(Order = 6)]
public sealed class WeatherSystem : ISystem
{
    private int _lastGameHour = -1;

    public void Update(World world)
    {
        var weather = world.Resource<WeatherCurrent>();
        var source = world.Resource<WeatherDataSource>();
        var state = world.Resource<GameState>();

        if (source.HourlyData.Count == 0) return;

        // Вычисляем игровое время
        DateTime gameTime = state.GameStartDate.AddSeconds(state.GameTimeElapsed);

        // Индекс часа = час суток (0-23), по модулю количества записей в CSV
        int gameHour = gameTime.Hour % source.HourlyData.Count;

        if (gameHour != _lastGameHour)
        {
            weather.Current = source.HourlyData[gameHour];
            weather.HourIndex = gameHour;
            weather.HasData = true;
            weather.GameTime = gameTime;
            _lastGameHour = gameHour;
        }
    }
}.
