using AsciiFarm.Content;
using AsciiFarm.Core;
using AsciiFarm.Data;
using AsciiFarm.Input;
using AsciiFarm.Rendering;
using AsciiFarm.Serialization;
using AsciiFarm.Systems;
using System.Collections.Concurrent;
using System.Reflection;

namespace AsciiFarm.App;

public sealed class GameApplication
{
    private readonly GameSettings _settings;
    public GameApplication(GameSettings settings) => _settings = settings;

    public void Run()
    {
        if (Console.IsInputRedirected)
            throw new InvalidOperationException("Нужна настоящая консоль: ввод перенаправлен.");

        try
        {
            Console.BufferWidth = _settings.ScreenWidth;
            Console.BufferHeight = _settings.ScreenHeight;
            Console.WindowWidth = _settings.ScreenWidth;
            Console.WindowHeight = _settings.ScreenHeight;
        }
        catch { /* остаёмся на текущем размере */ }

        int actualWidth = Console.WindowWidth;
        int actualHeight = Console.WindowHeight;

        var world = GameWorldBuilder.Build(_settings);
        var channel = new CommandChannel(_settings.CommandQueueCapacity);
        var frames = new FrameExchange<RenderFrame>(() => new RenderFrame(actualWidth, actualHeight));

        using var console = new DoubleBufferedConsole(actualWidth, actualHeight);
        using var shutdown = new CancellationTokenSource();
        var errors = new ConcurrentQueue<Exception>();

        KeyBindings bindings = _settings.UseJsonContent && Directory.Exists(_settings.ContentRoot)
            ? new ContentLoader(_settings.ContentRoot).LoadKeyBindings()
            : GameContent.CreateKeyBindings();

        var snapshot = new SnapshotSystem(frames, new MapRenderer(), new EntityRenderer(), new HudRenderer());

        var simulation = new SimulationLoop(
            world, channel, SystemPipeline.CreateSimulationSystems(),
            snapshot, frames, shutdown, _settings.SimulationFps);

        var fpsCounter = world.Resource<FpsCounter>();
        var render = new RenderLoop(console, frames,
            new ConsolePresentSystem(console, frames, fpsCounter),
            1000 / _settings.RenderFps);

        var input = new InputLoop(channel, bindings, _settings.InputPollMs);

        var threads = new[]
        {
            Start("Simulation", () => simulation.Run(shutdown.Token), shutdown, errors),
            Start("Render",     () => render.Run(shutdown.Token),     shutdown, errors),
            Start("Input",      () => input.Run(shutdown.Token),      shutdown, errors),
        };
        foreach (var thread in threads) thread.Join();

        channel.Close();
        if (!errors.IsEmpty) throw new AggregateException("Сбой игрового потока.", errors);

        world.Guard.Detach();
        if (world.Resource<GameState>().NeedsSave) SaveGame(world);
    }

    private void SaveGame(World world)
    {
        var registry = ComponentSerializerRegistry.AutoRegister(Assembly.GetExecutingAssembly());
        new ComponentSaveSystem(registry).Save(SafePath.Resolve(_settings.MapPath), world);
    }

    private static Thread Start(string name, Action body,
                                CancellationTokenSource shutdown, ConcurrentQueue<Exception> errors)
    {
        var thread = new Thread(() =>
        {
            try { body(); }
            catch (Exception ex) { errors.Enqueue(ex); shutdown.Cancel(); }
        })
        { Name = name, IsBackground = true };
        thread.Start();
        return thread;
    }
}.

using AsciiFarm.Components;
using AsciiFarm.Content;
using AsciiFarm.Core;
using AsciiFarm.Data;
using AsciiFarm.Serialization;
using AsciiFarm.Systems;
using System.Reflection;

namespace AsciiFarm.App;

public static class GameWorldBuilder
{
    public static World Build(GameSettings settings)
    {
        var world = new World();

        // P2-17: все пути проходят песочницу рабочей директории
        string mapPath = SafePath.Resolve(settings.MapPath);
        string weatherPath = SafePath.Resolve(settings.WeatherCsvPath);

        var registry = ComponentSerializerRegistry.AutoRegister(Assembly.GetExecutingAssembly());
        var saveSystem = new ComponentSaveSystem(registry);

        if (ComponentSaveSystem.IsNewFormat(mapPath))
        {
            saveSystem.Load(mapPath, world);
        }
        else if (File.Exists(mapPath))
        {
            // МИГРАЦИЯ: старый формат -> новый
            var legacyData = new GameStorage().LoadLatest<GameDataV0_0_3>(mapPath);

            var map = world.CreateEntity();
            world.Add(map, new TileMapComponent { Tiles = MigrateOldMap(legacyData.Map) });

            var player = world.CreateEntity();
            world.Add(player, new Position(legacyData.PlayerX, legacyData.PlayerY));
            world.Add(player, new PlayerTag());
            world.Add(player, new Glyph
            {
                Symbol = Cp437Symbols.WhiteSmilingFace,
                Foreground = ConsoleColor.Yellow,
                Background = ConsoleColor.Black,
                Layer = 10
            });

            saveSystem.Save(mapPath, world);
        }
        else
        {
            throw new FileNotFoundException($"Файл сохранения не найден: {mapPath}");
        }

        // --- Контент ---
        TilePalette palette;
        CharacterPalette charPalette;
        if (settings.UseJsonContent && Directory.Exists(settings.ContentRoot))
            (palette, _, charPalette) = GameContent.LoadFromFiles(settings.ContentRoot);
        else
        {
            palette = GameContent.CreatePalette();
            charPalette = GameContent.CreateCharacterPalette();
        }

        world.SetResource(palette);
        world.SetResource(charPalette);
        world.SetResource(new CommandBuffer());
        world.SetResource(new FpsCounter());
        world.SetResource(new FpsLimiter(settings.SimulationFps));
        world.SetResource(settings);

        // --- P2-10: состояние систем живёт в ресурсах ---
        world.SetResource(new SimulationClock());
        world.SetResource(new WeatherSystemState());
        world.SetResource(new WaterColorCache());

        // --- Погода (P0-1: ровно одна регистрация каждого ресурса) ---
        var weatherData = WeatherCsvLoader.Load(weatherPath, out var weatherError);
        if (settings.DaysPerMonth != 0)
            weatherData = WeatherCsvLoader.CompressWeatherData(weatherData, settings.DaysPerMonth);

        int startIndex = ComputeStartIndex(settings, weatherData.Count);
        world.SetResource(new WeatherDataSource
        {
            HourlyData = weatherData,
            LoadError = weatherError,
            StartIndex = startIndex
        });
        world.SetResource(new WeatherCurrent());

        world.SetResource(new GameState
        {
            GameStartDate = settings.GameStartDate,
            TimeSpeed = settings.DefaultTimeSpeed,
            DefaultTimeSpeed = settings.DefaultTimeSpeed,
            MaxTimeStep = settings.MaxTimeStep,
            DaysPerMonth = settings.DaysPerMonth
        });

        // --- Провайдеры информации о тайле (P0-1: каждый ровно один раз) ---
        var tileInfoRegistry = new TileInfoRegistry();
        tileInfoRegistry.Register(new TerrainInfoProvider());
        tileInfoRegistry.Register(new EntityInspectorProvider());
        tileInfoRegistry.Register(new WeatherTileInfoProvider());
        world.SetResource(tileInfoRegistry);

        return world;   // P0-1: единственный return
    }

    private static int ComputeStartIndex(GameSettings settings, int dataCount)
    {
        if (dataCount == 0) return 0;

        int startIndex;
        if (settings.DaysPerMonth == 0)
        {
            startIndex = (settings.GameStartDate.DayOfYear - 1) * 24 + settings.GameStartDate.Hour;
        }
        else
        {
            int month = Math.Clamp(settings.GameStartDate.Month, 1, 12);
            int day = Math.Clamp(settings.GameStartDate.Day, 1, settings.DaysPerMonth);
            int dayOfYear = (month - 1) * settings.DaysPerMonth + (day - 1);
            startIndex = dayOfYear * 24 + settings.GameStartDate.Hour;
        }
        return Math.Clamp(startIndex, 0, dataCount - 1);
    }

    private static TileData[,] MigrateOldMap(char[,] oldMap)
    {
        int width = oldMap.GetLength(0), height = oldMap.GetLength(1);
        var newMap = new TileData[width, height];
        for (int y = 0; y < height; y++)
            for (int x = 0; x < width; x++)
            {
                char c = oldMap[x, y];
                newMap[x, y] = c switch
                {
                    '≈' => new TileData { Symbol = c, Moisture = 1.0f },
                    '♣' or '♠' => new TileData { Symbol = c, Nitrogen = 0.7f, Phosphorus = 0.5f, Potassium = 0.6f, Moisture = 0.7f },
                    '.' or ',' or '\'' or '`' or ';' => new TileData { Symbol = c, Nitrogen = 0.5f, Phosphorus = 0.3f, Potassium = 0.4f, Moisture = 0.6f },
                    '#' or '▒' or '┌' or '┐' or '└' or '┘' or '─' or '│' or '┬' or '┴' or '├' or '┤' or '┼' or '╪'
                        or '╔' or '╗' or '╚' or '╝' or '═' or '║' or '╠' or '╣' or '╦' or '╩' or '╬'
                        => new TileData { Symbol = c },
                    ' ' => new TileData { Symbol = c, Nitrogen = 0.1f, Phosphorus = 0.1f, Potassium = 0.1f, Moisture = 0.1f },
                    _ => new TileData { Symbol = c, Nitrogen = 0.3f, Phosphorus = 0.3f, Potassium = 0.3f, Moisture = 0.4f }
                };
            }
        return newMap;
    }
}.

using AsciiFarm.Rendering;
using AsciiFarm.Systems;

namespace AsciiFarm.App;

public sealed class RenderLoop
{
    private readonly DoubleBufferedConsole _console;
    private readonly FrameExchange<RenderFrame> _frames;
    private readonly ConsolePresentSystem _present;
    private readonly int _frameMs;

    // P1-6: никаких Console.Error.WriteLine — альтернативный буфер этого не прощает.
    public RenderLoop(DoubleBufferedConsole console, FrameExchange<RenderFrame> frames,
                      ConsolePresentSystem present, int frameMs)
    {
        _console = console;
        _frames = frames;
        _present = present;
        _frameMs = frameMs;
    }

    public void Run(CancellationToken token)
    {
        _console.AttachToCurrentThread();
        _frames.AttachReader();
        var (lastWidth, lastHeight) = _console.GetWindowSize();

        while (!token.IsCancellationRequested)
        {
            var (w, h) = _console.GetWindowSize();
            if (w != lastWidth || h != lastHeight)
            {
                lastWidth = w; lastHeight = h;
                int nw = w, nh = h;
                _console.Resize(nw, nh);
                _frames.Resize(() => new RenderFrame(nw, nh));   // применит писатель
            }

            _present.Present();
            Thread.Sleep(_frameMs);
        }
    }
}.

using AsciiFarm.Core;
using AsciiFarm.Data;
using AsciiFarm.Input;
using AsciiFarm.Rendering;
using AsciiFarm.Systems;
namespace AsciiFarm.App;

public sealed class SimulationLoop
{
    private readonly World _world;
    private readonly CommandChannel _channel;
    private readonly ISystem[] _systems;
    private readonly SnapshotSystem _snapshot;
    private readonly FrameExchange<RenderFrame> _frames;
    private readonly CancellationTokenSource _shutdown;
    private readonly int _baseSimulationFps;
    private const int MaxTps = 1000;

    public SimulationLoop(World world, CommandChannel channel, ISystem[] systems,
                          SnapshotSystem snapshot, FrameExchange<RenderFrame> frames,
                          CancellationTokenSource shutdown, int baseSimulationFps)
    {
        _world = world; _channel = channel; _systems = systems;
        _snapshot = snapshot; _frames = frames; _shutdown = shutdown;
        _baseSimulationFps = baseSimulationFps;
    }

    public void Run(CancellationToken token)
    {
        _world.Guard.AttachToCurrentThread();
        _frames.AttachWriter();

        var state = _world.Resource<GameState>();
        var buffer = _world.Resource<CommandBuffer>();
        var limiter = _world.Resource<FpsLimiter>();

        while (!token.IsCancellationRequested && state.IsRunning)
        {
            // App/SimulationLoop.cs — фрагмент Run(), ветка промотки
            if (state.FastForwardUntil.HasValue)
            {
                while (state.GameTimeElapsed < state.FastForwardUntil.Value
                       && !token.IsCancellationRequested
                       && state.IsRunning)
                {
                    double remaining = state.FastForwardUntil.Value - state.GameTimeElapsed;
                    double step = Math.Min(remaining, state.MaxTimeStep);
                    state.GameTimeElapsed += step;   // только игровое время
                                                     // P0-2: Tick инкрементирует ТОЛЬКО ElapsedTimeSystem

                    using (_world.Guard.Enter())
                    {
                        _channel.DrainTo(buffer.Items);
                        foreach (var system in _systems) system.Update(_world);
                        _snapshot.Update(_world);
                    }
                }
                state.FastForwardUntil = null;
                continue;
            }

            // Обычный режим: динамический TPS
            int targetTps = Math.Max(_baseSimulationFps,
                (int)Math.Ceiling(state.TimeSpeed / state.MaxTimeStep));
            targetTps = Math.Min(targetTps, MaxTps);
            limiter.SetTargetFps(targetTps);

            while (!limiter.BeginTick())
            {
                Thread.Sleep(1);
                if (token.IsCancellationRequested || !state.IsRunning) break;
            }
            if (token.IsCancellationRequested || !state.IsRunning) break;

            using (_world.Guard.Enter())
            {
                _channel.DrainTo(buffer.Items);
                foreach (var system in _systems) system.Update(_world);
                _snapshot.Update(_world);
            }
        }

        _shutdown.Cancel();
    }
}.

    public static List<WeatherData> Load(string filePath, out string? error)
    {
        error = null;
        var result = new List<WeatherData>();
        if (!File.Exists(filePath)) { error = $"Файл не найден: {filePath}"; return result; }

        string[] lines;
        try { lines = File.ReadAllLines(filePath); }
        catch (Exception ex) { error = $"Не удалось прочитать: {ex.Message}"; return result; }

        if (lines.Length < 2) { error = "Файл слишком короткий"; return result; }

        int headerIndex = -1;
        for (int i = 0; i < Math.Min(10, lines.Length); i++)
        {
            var lower = lines[i].ToLowerInvariant();
            if (lower.Contains("time") && lower.Contains("temperature"))
            { headerIndex = i; break; }
        }
        if (headerIndex < 0) { error = "Не найдена строка заголовков"; return result; }

        var headers = ParseCsvLine(lines[headerIndex]).Select(NormalizeHeader).ToList();
        int idxTime = headers.IndexOf("time");
        int idxTemp = headers.IndexOf("temperature_2m");
        int idxHum = headers.IndexOf("relative_humidity_2m");
        int idxRad = headers.IndexOf("shortwave_radiation");
        int idxSoil = headers.IndexOf("soil_moisture_7_to_28cm");
        int idxSnow = headers.IndexOf("snow_depth");
        int idxPrecip = headers.IndexOf("precipitation");

        if (idxTemp < 0) { error = $"Нет столбца temperature_2m"; return result; }

        for (int i = headerIndex + 1; i < lines.Length; i++)
        {
            if (string.IsNullOrWhiteSpace(lines[i])) continue;
            var fields = ParseCsvLine(lines[i]);
            if (fields.Count == 0 || !LooksLikeDate(fields[0])) continue;
            if (fields.Count <= idxTemp) continue;

            // Парсим время
            DateTime timestamp = DateTime.MinValue;
            if (idxTime >= 0 && idxTime < fields.Count)
                DateTime.TryParse(fields[idxTime], CultureInfo.InvariantCulture,
                    DateTimeStyles.RoundtripKind, out timestamp);

            float temp = ParseFloat(fields, idxTemp);
            float hum = ParseFloat(fields, idxHum);
            float rad = ParseFloat(fields, idxRad);
            float soil = ParseFloat(fields, idxSoil);
            float snow = ParseFloat(fields, idxSnow);
            float precip = idxPrecip >= 0 ? ParseFloat(fields, idxPrecip) : 0f;

            float solarEnergyMj = rad * 0.0036f;
            float soilMoisture = Math.Clamp((soil - 0.10f) / 0.28f, 0f, 1f);
            float effectiveSnowCm = (snow * 100f) * 0.5f;
            if (temp > 3f) effectiveSnowCm = 0f;
            float groundTemp = (temp < 0f && effectiveSnowCm > 0f)
                ? temp / (1f + 0.15f * effectiveSnowCm) : temp;
            float airHumidity = Math.Clamp(hum / 100f, 0f, 1f);

            result.Add(new WeatherData
            {
                Timestamp = timestamp,
                AirTemp = temp,
                GroundTemp = groundTemp,
                AirHumidity = airHumidity,
                SoilMoisture = soilMoisture,
                SolarEnergyMj = solarEnergyMj,
                EffectiveSnowCm = effectiveSnowCm,
                PrecipitationMm = precip
            });
        }

        if (result.Count == 0) error = "Ни одна строка данных не распознана";
        return result;
    }

    private static string NormalizeHeader(string raw)
    {
        var s = raw.Trim().Trim('"');
        int paren = s.IndexOf('(');
        if (paren >= 0) s = s.Substring(0, paren);
        return s.Trim().ToLowerInvariant();
    }

    {
        if (string.IsNullOrEmpty(first)) return false;
        char c0 = first[0];
        return c0 >= '0' && c0 <= '9';
    }

    private static List<string> ParseCsvLine(string line)
    {
        var result = new List<string>();
        bool inQuotes = false;
        var current = new StringBuilder();
        foreach (char c in line)
        {
            if (c == '"') inQuotes = !inQuotes;
            else if (c == ',' && !inQuotes) { result.Add(current.ToString()); current.Clear(); }
            else current.Append(c);
        }
        result.Add(current.ToString());
        return result;
    }

        return float.TryParse(fields[index], NumberStyles.Float,

    private static List<WeatherData> CompressSingleYear(List<WeatherData> yearData, int year, int daysPerMonth)
    {
        int newDays = CalendarHelper.GetDaysInYear(daysPerMonth);

        var byDay = yearData.GroupBy(d => d.Timestamp.Date)
                            .OrderBy(g => g.Key)
                            .ToList();

        int originalDays = byDay.Count;
        double compressionRatio = (double)originalDays / newDays;

        // Динамическое окно с гарантированным перекрытием
        int windowSize = (int)Math.Ceiling(compressionRatio) + 1;
        if (windowSize % 2 == 0) windowSize++; // Делаем нечётным для симметрии
        int windowRadius = windowSize / 2;

        var result = new List<WeatherData>();

        for (int targetDay = 0; targetDay < newDays; targetDay++)
        {
            // Центр окна: равномерно распределяем по исходному году
            int centerDay = (int)Math.Round(targetDay * compressionRatio);

            var rangeRecords = new List<WeatherData>();

            for (int offset = -windowRadius; offset <= windowRadius; offset++)
            {
                int dayIndex = centerDay + offset;

                // Заворот на границах года
                if (dayIndex < 0)
                    dayIndex = originalDays + dayIndex;
                else if (dayIndex >= originalDays)
                    dayIndex = dayIndex - originalDays;

                rangeRecords.AddRange(byDay[dayIndex]);
            }

            for (int hour = 0; hour < 24; hour++)
            {
                var hourRecords = rangeRecords.Where(d => d.Timestamp.Hour == hour).ToList();
                if (hourRecords.Count == 0) continue;

                var gameTimestamp = new DateTime(year, 1, 1).AddDays(targetDay).AddHours(hour);

                result.Add(new WeatherData
                {
                    Timestamp = gameTimestamp,
                    AirTemp = (float)hourRecords.Average(d => d.AirTemp),
                    GroundTemp = (float)hourRecords.Average(d => d.GroundTemp),
                    AirHumidity = (float)hourRecords.Average(d => d.AirHumidity),
                    SoilMoisture = (float)hourRecords.Average(d => d.SoilMoisture),
                    SolarEnergyMj = (float)hourRecords.Average(d => d.SolarEnergyMj),
                    EffectiveSnowCm = (float)hourRecords.Average(d => d.EffectiveSnowCm),
                    PrecipitationMm = (float)hourRecords.Average(d => d.PrecipitationMm)
                });
            }
        }

        return result;
    }

/// <summary>Только хранилище данных. Никакой игровой логики.</summary>
public sealed class ComponentStore<T> : IComponentStore where T : struct
{
    private readonly Dictionary<Entity, T> _items = new();

    public int Count => _items.Count;
    public int StructureVersion { get; private set; }
    public IReadOnlyCollection<Entity> Entities => _items.Keys;

    public bool Has(Entity entity) => _items.ContainsKey(entity);

    public void Set(Entity entity, T component)
    {
        if (!_items.ContainsKey(entity)) StructureVersion++;   // перезапись значения версию не меняет
        _items[entity] = component;
    }

    public T Get(Entity entity) => _items[entity];
    public bool TryGet(Entity entity, out T component) => _items.TryGetValue(entity, out component);

    public void Remove(Entity entity)
    {
        if (_items.Remove(entity)) StructureVersion++;
    }

    public void Accept(IComponentVisitor visitor)
    {
        foreach (var kvp in _items) visitor.Visit(kvp.Key, kvp.Value);
    }

    /// <summary>Альтернатива для простых случаев: делегат типизирован, компонент не упаковывается.</summary>
    public void Visit(Action<Entity, T> action)
    {
        foreach (var kvp in _items) action(kvp.Key, kvp.Value);
    }

public sealed class FpsLimiter
{
    private double _targetFrameTimeSeconds;
    private readonly Stopwatch _stopwatch = new();
    private double _lastAccumulatedTime;

    public int TargetFps { get; private set; }
    public float DeltaTime { get; private set; }
    public double ElapsedTime => _stopwatch.Elapsed.TotalSeconds;

    public FpsLimiter(int targetFps)
    {
        TargetFps = targetFps > 0 ? targetFps
            : throw new ArgumentOutOfRangeException(nameof(targetFps));
        _targetFrameTimeSeconds = 1.0 / targetFps;
    }

    /// <summary>Изменить целевой FPS на лету (для ускорения времени).</summary>
    public void SetTargetFps(int targetFps)
    {
        if (targetFps <= 0) return;
        TargetFps = targetFps;
        _targetFrameTimeSeconds = 1.0 / targetFps;
        // Сбрасываем накопитель, чтобы не было скачка
        _lastAccumulatedTime = _stopwatch.Elapsed.TotalSeconds;
    }

    public bool BeginTick()
    {
        if (!_stopwatch.IsRunning) _stopwatch.Start();

        double now = _stopwatch.Elapsed.TotalSeconds;
        double delta = now - _lastAccumulatedTime;

        if (delta < _targetFrameTimeSeconds * 0.95) return false;

        DeltaTime = (float)Math.Min(delta, _targetFrameTimeSeconds * 2);
        _lastAccumulatedTime = now;
        return true;
    }
}.

/// <summary>
/// Песочница путей: ни один файл не читается/пишется вне рабочей директории.
/// Корень по умолчанию — текущая директория (как и было у относительных путей).
/// Можно переопределить: SafePath.Root = AppContext.BaseDirectory.
/// </summary>
public static class SafePath
{
    private static readonly StringComparison Cmp =
        OperatingSystem.IsWindows() ? StringComparison.OrdinalIgnoreCase : StringComparison.Ordinal;

    private static string _root = Path.GetFullPath(Directory.GetCurrentDirectory());

    public static string Root
    {
        get => _root;
        set => _root = Path.GetFullPath(value);
    }

    public static string Resolve(string path)
    {
        if (string.IsNullOrWhiteSpace(path))
            throw new ArgumentException("Путь пуст.", nameof(path));
        if (path.IndexOfAny(Path.GetInvalidPathChars()) >= 0)
            throw new ArgumentException($"Недопустимые символы в пути: '{path}'.", nameof(path));

        string full = Path.GetFullPath(Path.IsPathRooted(path) ? path : Path.Combine(_root, path));
        string prefix = _root.EndsWith(Path.DirectorySeparatorChar) ? _root : _root + Path.DirectorySeparatorChar;

        if (!full.StartsWith(prefix, Cmp) && !string.Equals(full, _root, Cmp))
            throw new UnauthorizedAccessException(
                $"Путь '{path}' выходит за пределы рабочей директории '{_root}'.");

        return full;
    }
}.

namespace AsciiFarm.Core;

/// <summary>
/// Блокиратор потока-владельца. AttachToCurrentThread() вызывается в начале потока,
/// после чего любой доступ из другого потока — исключение, а не гонка.
/// </summary>
public sealed class ThreadGuard
{
    private readonly string _name;
    private int _ownerThreadId;   // 0 = ещё не привязан
    private int _busy;            // 1 = данные прямо сейчас модифицируются

    public ThreadGuard(string name) => _name = name;

    public void AttachToCurrentThread() =>
        Interlocked.Exchange(ref _ownerThreadId, Environment.CurrentManagedThreadId);

    public void Validate()
    {
        int owner = Volatile.Read(ref _ownerThreadId);
        if (owner != 0 && owner != Environment.CurrentManagedThreadId)
            throw new InvalidOperationException(
                $"«{_name}»: данные принадлежат потоку {owner}, " +
                $"доступ из потока {Environment.CurrentManagedThreadId} запрещён.");
    }

    /// <summary>Критическая секция изменения данных (запрещает повторный вход).</summary>
    public Scope Enter()
    {
        Validate();
        if (Interlocked.Exchange(ref _busy, 1) != 0)
            throw new InvalidOperationException($"«{_name}»: данные ещё модифицируются.");
        return new Scope(this);
    }

    private void Release() => Volatile.Write(ref _busy, 0);

    public readonly struct Scope : IDisposable
    {
        private readonly ThreadGuard _guard;
        internal Scope(ThreadGuard guard) => _guard = guard;
        public void Dispose() => _guard.Release();
    }
    /// <summary>
    /// Сбрасывает владельца и флаг занятости.
    /// Вызывать после завершения потока-владельца, если данные больше не модифицируются.
    /// </summary>
    public void Detach()
    {
        Volatile.Write(ref _busy, 0);
        Volatile.Write(ref _ownerThreadId, 0);
    }
}.

using System;
using System.Collections.Generic;
using System.Linq;

namespace AsciiFarm.Core;

/// <summary>
/// Данные мира. Принадлежат РОВНО ОДНОМУ потоку (симуляции).
/// ВНИМАНИЕ: массивы из Query&lt;...&gt; кэшируются — вызывающий код не должен их изменять.
/// Нужна мутабельная копия — делайте .ToArray() самостоятельно.
/// </summary>
public sealed class World
{
    private readonly Dictionary<Type, IComponentStore> _stores = new();
    private readonly Dictionary<Type, object> _resources = new();
    private readonly Dictionary<QueryKey, CachedQuery> _queryCache = new();
    private int _nextEntityId;
    private int _schemaVersion;   // растёт при появлении нового типа компонента

    public ThreadGuard Guard { get; } = new("World");

    public Entity CreateEntity()
    {
        Guard.Validate();
        return new Entity(_nextEntityId++);
    }

    public ComponentStore<T> Store<T>() where T : struct
    {
        Guard.Validate();
        if (!_stores.TryGetValue(typeof(T), out var store))
        {
            store = new ComponentStore<T>();
            _stores[typeof(T)] = store;
            _schemaVersion++;
            _queryCache.Clear();
        }
        return (ComponentStore<T>)store;
    }

    public void Add<T>(Entity entity, T component) where T : struct => Store<T>().Set(entity, component);

    public Entity[] Query<TA>() where TA : struct
    {
        IComponentStore a = Store<TA>();
        var key = new QueryKey(typeof(TA), null, null);
        if (TryGetCached(key, a, null, null, out var hit)) return hit;

        var result = a.Entities.ToArray();
        PutCache(key, a, null, null, result);
        return result;
    }

    public Entity[] Query<TA, TB>() where TA : struct where TB : struct
    {
        IComponentStore a = Store<TA>();
        IComponentStore b = Store<TB>();
        var key = new QueryKey(typeof(TA), typeof(TB), null);
        if (TryGetCached(key, a, b, null, out var hit)) return hit;

        IComponentStore small = a.Count <= b.Count ? a : b;
        IComponentStore large = a.Count <= b.Count ? b : a;

        var result = small.Entities.Where(large.Has).ToArray();
        PutCache(key, a, b, null, result);
        return result;
    }

    public Entity[] Query<TA, TB, TC>() where TA : struct where TB : struct where TC : struct
    {
        IComponentStore a = Store<TA>();
        IComponentStore b = Store<TB>();
        IComponentStore c = Store<TC>();
        var key = new QueryKey(typeof(TA), typeof(TB), typeof(TC));
        if (TryGetCached(key, a, b, c, out var hit)) return hit;

        IComponentStore small = a.Count <= b.Count ? a : b;
        IComponentStore large = a.Count <= b.Count ? b : a;

        var result = small.Entities.Where(e => large.Has(e) && c.Has(e)).ToArray();
        PutCache(key, a, b, c, result);
        return result;
    }

    public T Resource<T>() where T : class
    {
        Guard.Validate();
        return (T)_resources[typeof(T)];
    }

    /// <summary>Отсутствие ресурса — не исключение.</summary>
    public bool TryGetResource<T>(out T? resource) where T : class
    {
        Guard.Validate();
        if (_resources.TryGetValue(typeof(T), out var obj) && obj is T typed)
        {
            resource = typed;
            return true;
        }
        resource = null;
        return false;
    }

    public void SetResource<T>(T resource) where T : class
    {
        Guard.Validate();
        _resources[typeof(T)] = resource;
    }

    /// <summary>Итератор по всем хранилищам компонентов.</summary>
    public IEnumerable<(Type ComponentType, IComponentStore Store)> AllStores()
    {
        Guard.Validate();
        foreach (var kvp in _stores) yield return (kvp.Key, kvp.Value);
    }

    /// <summary>Резервирует ID, чтобы новые сущности не конфликтовали с загруженными.</summary>
    public void ReserveId(int id)
    {
        Guard.Validate();
        if (id >= _nextEntityId) _nextEntityId = id + 1;
    }

    // --- кэш запросов: инвалидация по структурным версиям хранилищ ---
    private bool TryGetCached(QueryKey key, IComponentStore a, IComponentStore? b, IComponentStore? c,
                              out Entity[] result)
    {
        if (_queryCache.TryGetValue(key, out var cached)
            && cached.Schema == _schemaVersion
            && cached.Va == a.StructureVersion
            && cached.Vb == (b?.StructureVersion ?? 0)
            && cached.Vc == (c?.StructureVersion ?? 0))
        {
            result = cached.Entities;
            return true;
        }
        result = Array.Empty<Entity>();
        return false;
    }

    private void PutCache(QueryKey key, IComponentStore a, IComponentStore? b, IComponentStore? c, Entity[] result) =>
        _queryCache[key] = new CachedQuery(_schemaVersion, a.StructureVersion,
                                           b?.StructureVersion ?? 0, c?.StructureVersion ?? 0, result);

    private readonly record struct QueryKey(Type A, Type? B, Type? C);
    private readonly record struct CachedQuery(int Schema, int Va, int Vb, int Vc, Entity[] Entities);
}.

using System;

namespace AsciiFarm.Data;

public static class CalendarHelper
{
    public const double SecondsPerMinute = 60.0;
    public const double SecondsPerHour = 3600.0;
    public const double SecondsPerDay = 86400.0;
    public const int MonthsPerYear = 12;

    /// <summary>Дата/время игрового календаря.</summary>
    public readonly record struct GameDate(int Year, int Month, int Day, int Hour, int Minute, int Second);

    /// <summary>Дней в игровом году. 0 = реальный календарь (365).</summary>
    public static int GetDaysInYear(int daysPerMonth) =>
        daysPerMonth > 0 ? daysPerMonth * MonthsPerYear : 365;

    /// <summary>Дней в игровом месяце. 0 = реальный календарь.</summary>
    public static int GetDaysInMonth(int year, int month, int daysPerMonth) =>
        daysPerMonth > 0
            ? daysPerMonth
            : DateTime.DaysInMonth(year, Math.Clamp(month, 1, 12));

    /// <summary>Текущая игровая дата с учётом кастомного календаря.</summary>
    public static GameDate GetGameDate(DateTime startDate, double gameSeconds, int daysPerMonth)
    {
        if (daysPerMonth <= 0)
        {
            var real = AddSecondsSafe(startDate, gameSeconds);
            return new GameDate(real.Year, real.Month, real.Day, real.Hour, real.Minute, real.Second);
        }

        int daysPerYear = daysPerMonth * MonthsPerYear;

        int startMonth = Math.Clamp(startDate.Month, 1, 12);
        int startDay = Math.Clamp(startDate.Day, 1, daysPerMonth);
        int startDayOfYear = (startMonth - 1) * daysPerMonth + (startDay - 1);   // 0-based

        long totalSeconds = (long)gameSeconds
                            + startDayOfYear * 86400L
                            + startDate.Hour * 3600L
                            + startDate.Minute * 60L
                            + startDate.Second;

        long totalDays = totalSeconds / 86400L;
        long remSeconds = totalSeconds % 86400L;
        long years = totalDays / daysPerYear;
        long remDays = totalDays % daysPerYear;

        return new GameDate(
            startDate.Year + (int)years,
            (int)(remDays / daysPerMonth) + 1,
            (int)(remDays % daysPerMonth) + 1,
            (int)(remSeconds / 3600),
            (int)(remSeconds % 3600 / 60),
            (int)(remSeconds % 60));
    }

    /// <summary>Длина игрового месяца в секундах (для P+4 / X+4).</summary>
    public static double GetMonthSeconds(DateTime startDate, double gameSeconds, int daysPerMonth)
    {
        if (daysPerMonth > 0) return daysPerMonth * SecondsPerDay;
        var d = GetGameDate(startDate, gameSeconds, daysPerMonth);
        return DateTime.DaysInMonth(d.Year, d.Month) * SecondsPerDay;
    }

    /// <summary>Длина игрового года в секундах: 12 игровых месяцев (для P+5 / X+5).</summary>
    public static double GetYearSeconds(DateTime startDate, double gameSeconds, int daysPerMonth)
    {
        if (daysPerMonth > 0) return daysPerMonth * MonthsPerYear * SecondsPerDay;
        var d = GetGameDate(startDate, gameSeconds, daysPerMonth);
        return (DateTime.IsLeapYear(d.Year) ? 366 : 365) * SecondsPerDay;
    }

    /// <summary>Форматирует игровое время с учётом кастомного календаря.</summary>
    public static string FormatGameTime(DateTime startDate, double gameSeconds, int daysPerMonth)
    {
        var d = GetGameDate(startDate, gameSeconds, daysPerMonth);
        return $"{d.Day:D2}.{d.Month:D2}.{d.Year:D4} {d.Hour:D2}:{d.Minute:D2}:{d.Second:D2}";
    }

    // Защита от выхода за DateTime.MaxValue при длинных промотках в режиме реального календаря.
    private static DateTime AddSecondsSafe(DateTime start, double seconds)
    {
        double max = (DateTime.MaxValue - start).TotalSeconds;
        return start.AddSeconds(Math.Clamp(seconds, 0.0, max));
    }
}.

/// <summary>Потокобезопасный счётчик: RecordFrame — render-поток, RecordTick — симуляция.</summary>
public sealed class FpsCounter
{
    private const int SampleSize = 60;

    private readonly object _gate = new();
    private readonly Queue<double> _frameTimes = new();
    private readonly Stopwatch _renderTimer = Stopwatch.StartNew();
    private readonly Stopwatch _tickTimer = Stopwatch.StartNew();

    private int _framesThisSecond;
    private int _ticksThisSecond;
    private float _currentFps;
    private float _averageFps;
    private int _realFps;
    private int _tps;

    public float CurrentFps { get { lock (_gate) return _currentFps; } }
    public float AverageFps { get { lock (_gate) return _averageFps; } }
    public int RealFps { get { lock (_gate) return _realFps; } }
    public int TicksPerSecond { get { lock (_gate) return _tps; } }

    /// <param name="timestampSeconds">Монотонное реальное время (Stopwatch), НЕ номер тика.</param>
    public void RecordFrame(double timestampSeconds)
    {
        lock (_gate)
        {
            _frameTimes.Enqueue(timestampSeconds);
            if (_frameTimes.Count > SampleSize) _frameTimes.Dequeue();

            if (_frameTimes.Count >= 2)
            {
                double duration = timestampSeconds - _frameTimes.Peek();   // без ToArray()
                if (duration > 0.001)
                    _currentFps = (float)((_frameTimes.Count - 1) / duration);
            }

            _framesThisSecond++;
            if (_renderTimer.ElapsedMilliseconds >= 1000)
            {
                _realFps = _framesThisSecond;
                _framesThisSecond = 0;
                _renderTimer.Restart();
            }
        }
    }

    public void RecordTick()
    {
        lock (_gate)
        {
            _ticksThisSecond++;
            if (_tickTimer.ElapsedMilliseconds >= 1000)
            {
                _tps = _ticksThisSecond;
                _ticksThisSecond = 0;
                _tickTimer.Restart();
            }
        }
    }

    public void UpdateAverage(float instantFps)
    {
        lock (_gate)
            _averageFps = _averageFps == 0 ? instantFps : _averageFps * 0.9f + instantFps * 0.1f;
    }
}.

/// <summary>
/// АВТОМАТИЧЕСКИЙ ИНСПЕКТОР: находит ВСЕ сущности на данных координатах

public sealed class EntityInspectorProvider : ITileInfoProvider
{
    public IEnumerable<string> GetInfo(int x, int y, char tile, World world)
    {
        var positions = world.Store<Position>();

        foreach (var entity in positions.Entities)
        {
            var pos = positions.Get(entity);
            if (pos.X != x || pos.Y != y) continue;

            yield return $"── Объект (ID: {entity.Id}) ──";

            var visitor = new InspectorVisitor(entity);

        {
            if (entity.Id != _targetId) return;
            if (typeof(T) == typeof(Position)) return;   // координаты уже показаны
            Lines.Add($"  {typeof(T).Name}: {ComponentDisplayRegistry.Format(in component)}");
        }
    }

namespace AsciiFarm.Input;

public sealed class CommandChannel
{
    private readonly Queue<GameCommand> _queue = new();
    private readonly object _gate = new();
    private readonly int _capacity;
    private bool _closed;

    public CommandChannel(int capacity) => _capacity = capacity;

    public void Post(GameCommand command)
    {
        lock (_gate)
        {
            if (_closed || command == GameCommand.None) return;
            if (_queue.Count >= _capacity) _queue.Dequeue();
            _queue.Enqueue(command);
        }
    }

    public void DrainTo(List<GameCommand> target)
    {
        lock (_gate)
        {
            while (_queue.Count > 0) target.Add(_queue.Dequeue());
        }
    }

    public void Close()
    {
        lock (_gate) { _closed = true; _queue.Clear(); }
    }
}.

using AsciiFarm.Data;

namespace AsciiFarm.Input;

public sealed class InputLoop
{
    private enum Modifier { None, Speed, FastForward }

    private readonly CommandChannel _channel;
    private readonly KeyBindings _bindings;
    private readonly int _pollDelayMs;
    private Modifier _lastModifier = Modifier.None;

    public InputLoop(CommandChannel channel, KeyBindings bindings, int pollDelayMs)
    {
        _channel = channel;
        _bindings = bindings;
        _pollDelayMs = pollDelayMs;
    }

    public void Run(CancellationToken token)
    {
        while (!token.IsCancellationRequested)
        {
            bool worked = false;
            while (Console.KeyAvailable)
            {
                var key = Console.ReadKey(intercept: true).Key;

                if (key == _bindings.SpeedModifier)
                { _lastModifier = Modifier.Speed; worked = true; continue; }

                if (key == _bindings.FastForwardModifier)
                { _lastModifier = Modifier.FastForward; worked = true; continue; }

                if (key >= ConsoleKey.D1 && key <= ConsoleKey.D5 && _lastModifier != Modifier.None)
                {
                    int n = key - ConsoleKey.D1 + 1;
                    _channel.Post(_lastModifier == Modifier.Speed ? SpeedCommand(n) : ForwardCommand(n));
                    _lastModifier = Modifier.None;
                    worked = true;
                    continue;
                }

                _lastModifier = Modifier.None;   // любая другая клавиша сбрасывает модификатор

                if (_bindings.Map.TryGetValue(key, out var command))
                { _channel.Post(command); worked = true; }
            }
            if (!worked) Thread.Sleep(_pollDelayMs);
        }
    }

    private static GameCommand SpeedCommand(int n) => n switch
    {
        1 => GameCommand.Speed1,
        2 => GameCommand.Speed2,
        3 => GameCommand.Speed3,
        4 => GameCommand.Speed4,
        5 => GameCommand.Speed5,
        _ => GameCommand.None
    };

    private static GameCommand ForwardCommand(int n) => n switch
    {
        1 => GameCommand.FastForward1,
        2 => GameCommand.FastForward2,
        3 => GameCommand.FastForward3,
        4 => GameCommand.FastForward4,
        5 => GameCommand.FastForward5,
        _ => GameCommand.None
    };
}.

﻿using System;
using System.ComponentModel;
using System.Runtime.InteropServices;
using Microsoft.Win32.SafeHandles;

namespace AsciiFarm.Rendering;

public sealed class DoubleBufferedConsole : IDisposable
{
    private const int STD_OUTPUT_HANDLE = -11;
    private const uint GENERIC_READ = 0x80000000;
    private const uint GENERIC_WRITE = 0x40000000;
    private const uint FILE_SHARE_READ = 0x00000001;
    private const uint FILE_SHARE_WRITE = 0x00000002;
    private const uint CONSOLE_TEXTMODE_BUFFER = 1;

    private readonly AsciiFarm.Core.ThreadGuard _guard = new("Console");
    public void AttachToCurrentThread() => _guard.AttachToCurrentThread();

    [StructLayout(LayoutKind.Sequential)]
    private struct COORD { public short X; public short Y; }

    [StructLayout(LayoutKind.Sequential)]
    private struct SMALL_RECT
    {
        public short Left;
        public short Top;
        public short Right;
        public short Bottom;
    }

    [StructLayout(LayoutKind.Explicit, CharSet = CharSet.Unicode)]
    private struct CHAR_INFO
    {
        [FieldOffset(0)] public char UnicodeChar;
        [FieldOffset(0)] public byte AsciiChar;
        [FieldOffset(2)] public short Attributes;
    }

    [StructLayout(LayoutKind.Sequential)]
    private struct CONSOLE_SCREEN_BUFFER_INFO
    {
        public COORD dwSize;
        public COORD dwCursorPosition;
        public short wAttributes;
        public SMALL_RECT srWindow;
        public COORD dwMaximumWindowSize;
    }

    [DllImport("kernel32.dll", SetLastError = true)]
    private static extern nint GetStdHandle(int nStdHandle);

    [DllImport("kernel32.dll", SetLastError = true)]
    private static extern COORD GetLargestConsoleWindowSize(SafeFileHandle hConsoleOutput);

    [DllImport("kernel32.dll", SetLastError = true)]
    [return: MarshalAs(UnmanagedType.Bool)]
    private static extern bool SetConsoleScreenBufferSize(SafeFileHandle hConsoleOutput, COORD dwSize);

    [DllImport("kernel32.dll", SetLastError = true)]
    [return: MarshalAs(UnmanagedType.Bool)]
    private static extern bool SetConsoleWindowInfo(
        SafeFileHandle hConsoleOutput,
        [MarshalAs(UnmanagedType.Bool)] bool bAbsolute,
        ref SMALL_RECT lpConsoleWindow);

    [DllImport("kernel32.dll", SetLastError = true)]
    [return: MarshalAs(UnmanagedType.Bool)]
    private static extern bool GetConsoleScreenBufferInfo(
        SafeFileHandle hConsoleOutput,
        out CONSOLE_SCREEN_BUFFER_INFO lpConsoleScreenBufferInfo);

    [DllImport("kernel32.dll", SetLastError = true, CharSet = CharSet.Unicode)]
    [return: MarshalAs(UnmanagedType.Bool)]
    private static extern bool WriteConsoleOutput(
        SafeFileHandle hConsoleOutput,
        CHAR_INFO[] lpBuffer,
        COORD dwBufferSize,
        COORD dwBufferCoord,
        ref SMALL_RECT lpConsoleWriteRegion);

    [DllImport("kernel32.dll", SetLastError = true)]
    private static extern SafeFileHandle CreateConsoleScreenBuffer(
        uint dwDesiredAccess,
        uint dwShareMode,
        nint lpSecurityAttributes,
        uint dwFlags,
        nint lpScreenBufferData);

    [DllImport("kernel32.dll", SetLastError = true)]
    [return: MarshalAs(UnmanagedType.Bool)]
    private static extern bool SetConsoleActiveScreenBuffer(SafeFileHandle hConsoleOutput);

    private readonly SafeFileHandle _stdout;
    private SafeFileHandle _buffer;
    private CHAR_INFO[] _cells;
    private int _width;
    private int _height;
    private bool _dirty;
    private bool _disposed;

    public int Width => _width;
    public int Height => _height;
    public ConsoleColor DefaultForeground { get; set; }
    public ConsoleColor DefaultBackground { get; set; }

    public DoubleBufferedConsole(
        int width, int height,
        ConsoleColor defaultForeground = ConsoleColor.Gray,
        ConsoleColor defaultBackground = ConsoleColor.Black)
    {
        if (!RuntimeInformation.IsOSPlatform(OSPlatform.Windows))
            throw new PlatformNotSupportedException("Поддерживается только Windows.");
        if (Console.IsOutputRedirected)
            throw new InvalidOperationException("Вывод консоли перенаправлен. Нужна настоящая консоль.");

        ValidateSize(width, height);
        _width = width;
        _height = height;
        DefaultForeground = defaultForeground;
        DefaultBackground = defaultBackground;

        _stdout = OpenStandardOutput();

        try
        {
            ValidateAgainstLargestWindowSize(width, height);
            _buffer = CreateBuffer();

            // КЛЮЧЕВОЕ ИСПРАВЛЕНИЕ: буфер должен быть активным ДО работы с его окном.
            // SetConsoleWindowInfo на неактивном буфере возвращает ERROR_INVALID_PARAMETER.
            SetActiveBuffer(_buffer);

            SetBufferSize(width, height);

            _cells = new CHAR_INFO[width * height];
            FillCells(_cells, ' ', MakeAttribute(DefaultForeground, DefaultBackground));
            WriteBuffer(_buffer, _cells, width, height);
            _dirty = false;

            try { Console.CursorVisible = false; } catch { }
        }
        catch
        {
            _buffer?.Dispose();
            _stdout.Dispose();
            throw;
        }
    }

    public void PutChar(int x, int y, char character, ConsoleColor foreground, ConsoleColor background)
    {
        _guard.Validate();
        CheckNotDisposed();
        if ((uint)x >= (uint)_width || (uint)y >= (uint)_height) return;
        int index = y * _width + x;
        _cells[index].UnicodeChar = character;
        _cells[index].Attributes = MakeAttribute(foreground, background);
        _dirty = true;
    }

    public void Draw(bool force = false)
    {
        _guard.Validate();
        CheckNotDisposed();
        if (!_dirty && !force) return;
        WriteBuffer(_buffer, _cells, _width, _height);
        _dirty = false;
    }

    public void Clear()
    {
        _guard.Validate();
        CheckNotDisposed();
        Fill(' ', DefaultForeground, DefaultBackground);
    }

    public void Fill(char character, ConsoleColor foreground, ConsoleColor background)
    {
        _guard.Validate();
        CheckNotDisposed();
        short attributes = MakeAttribute(foreground, background);
        FillCells(_cells, character, attributes);
        _dirty = true;
    }

    public void FillRect(int x, int y, int width, int height, char character,
        ConsoleColor foreground, ConsoleColor background)
    {
        _guard.Validate();
        CheckNotDisposed();
        if (width < 0 || height < 0)
            throw new ArgumentOutOfRangeException(nameof(width), "Размер не может быть отрицательным.");
        if (width == 0 || height == 0) return;

        int left = Math.Max(0, x);
        int top = Math.Max(0, y);
        int right = (int)Math.Min(_width, (long)x + width);
        int bottom = (int)Math.Min(_height, (long)y + height);
        if (left >= right || top >= bottom) return;

        short attributes = MakeAttribute(foreground, background);
        for (int yy = top; yy < bottom; yy++)
        {
            int rowStart = yy * _width;
            for (int xx = left; xx < right; xx++)
            {
                int index = rowStart + xx;
                _cells[index].UnicodeChar = character;
                _cells[index].Attributes = attributes;
            }
        }
        _dirty = true;
    }

    public void Resize(int width, int height)
    {
        _guard.Validate();
        CheckNotDisposed();
        ValidateSize(width, height);
        if (width == _width && height == _height) return;
        ValidateAgainstLargestWindowSize(width, height);

        SetBufferSize(width, height);

        _cells = new CHAR_INFO[width * height];
        FillCells(_cells, ' ', MakeAttribute(DefaultForeground, DefaultBackground));
        _width = width;
        _height = height;

        // Сразу затираем весь буфер пробелами, не дожидаясь свежего кадра.
        // Это убирает мусор, который conhost оставляет в новых ячейках при растяжении.
        WriteBuffer(_buffer, _cells, width, height);
        _dirty = false;
    }

    public (int Width, int Height) GetWindowSize()
    {
        _guard.Validate();
        CheckNotDisposed();
        if (GetConsoleScreenBufferInfo(_buffer, out var info))
        {
            int w = info.srWindow.Right - info.srWindow.Left + 1;
            int h = info.srWindow.Bottom - info.srWindow.Top + 1;
            return (w, h);
        }
        return (_width, _height);
    }

    public void Dispose()
    {
        if (_disposed) return;
        _disposed = true;

        try
        {
            if (_stdout != null && !_stdout.IsInvalid)
                SetConsoleActiveScreenBuffer(_stdout);
        }
        catch { }

        _buffer?.Dispose();
        _stdout?.Dispose();

        try { Console.CursorVisible = true; } catch { }
    }

    private void CheckNotDisposed()
    {
        if (_disposed) throw new ObjectDisposedException(nameof(DoubleBufferedConsole));
    }

    private static void ValidateSize(int width, int height)
    {
        if (width <= 0 || height <= 0)
            throw new ArgumentOutOfRangeException(nameof(width), "Размеры должны быть положительными.");
        if (width > short.MaxValue || height > short.MaxValue)
            throw new ArgumentOutOfRangeException(nameof(width), "Размеры слишком большие для COORD.");
        if ((long)width * height > int.MaxValue)
            throw new ArgumentOutOfRangeException(nameof(width), "Слишком много ячеек.");
    }

    private void ValidateAgainstLargestWindowSize(int width, int height)
    {
        COORD largest = GetLargestConsoleWindowSize(_stdout);
        if (largest.X > 0 && largest.Y > 0 && (width > largest.X || height > largest.Y))
            throw new ArgumentOutOfRangeException(
                nameof(width),
                $"Размер окна консоли не может быть больше {largest.X}x{largest.Y}.");
    }

    private static SafeFileHandle OpenStandardOutput()
    {
        SafeFileHandle handle = new SafeFileHandle(GetStdHandle(STD_OUTPUT_HANDLE), false);
        if (handle.IsInvalid) throw new Win32Exception(Marshal.GetLastWin32Error());
        return handle;
    }

    private SafeFileHandle CreateBuffer()
    {
        SafeFileHandle handle = CreateConsoleScreenBuffer(
            GENERIC_READ | GENERIC_WRITE,
            FILE_SHARE_READ | FILE_SHARE_WRITE,
            nint.Zero,
            CONSOLE_TEXTMODE_BUFFER,
            nint.Zero);
        if (handle.IsInvalid) throw new Win32Exception(Marshal.GetLastWin32Error());
        return handle;
    }

    /// <summary>
    /// Устанавливает размер буфера и окна.
    /// Соблюдает инвариант Windows: размер окна не может превышать размер буфера.
    /// Никаких «трюков» с 1×1 — только корректная последовательность.
    /// </summary>
    private void SetBufferSize(int width, int height)
    {
        if (!GetConsoleScreenBufferInfo(_buffer, out var info))
            throw new Win32Exception(Marshal.GetLastWin32Error());

        int curBufW = info.dwSize.X;
        int curBufH = info.dwSize.Y;
        int curWinW = info.srWindow.Right - info.srWindow.Left + 1;
        int curWinH = info.srWindow.Bottom - info.srWindow.Top + 1;

        COORD size = new COORD { X = (short)width, Y = (short)height };
        SMALL_RECT target = new SMALL_RECT
        {
            Left = 0,
            Top = 0,
            Right = (short)(width - 1),
            Bottom = (short)(height - 1)
        };

        // Если уменьшаем буфер, а текущее окно больше нового буфера —
        // сначала уменьшаем окно до целевого размера (это всегда разрешено).
        if (width < curWinW || height < curWinH)
        {
            SMALL_RECT interim = target;
            if (width > curBufW) interim.Right = (short)(curBufW - 1);
            if (height > curBufH) interim.Bottom = (short)(curBufH - 1);
            if (!SetConsoleWindowInfo(_buffer, true, ref interim))
                throw new Win32Exception(Marshal.GetLastWin32Error());
        }

        // Меняем размер буфера.
        if (!SetConsoleScreenBufferSize(_buffer, size))
            throw new Win32Exception(Marshal.GetLastWin32Error());

        // Устанавливаем финальный размер окна.
        if (!SetConsoleWindowInfo(_buffer, true, ref target))
            throw new Win32Exception(Marshal.GetLastWin32Error());
    }

    private static void WriteBuffer(SafeFileHandle handle, CHAR_INFO[] cells, int width, int height)
    {
        COORD bufferSize = new COORD { X = (short)width, Y = (short)height };
        COORD bufferCoord = new COORD { X = 0, Y = 0 };
        SMALL_RECT region = new SMALL_RECT
        {
            Left = 0,
            Top = 0,
            Right = (short)(width - 1),
            Bottom = (short)(height - 1)
        };
        if (!WriteConsoleOutput(handle, cells, bufferSize, bufferCoord, ref region))
            throw new Win32Exception(Marshal.GetLastWin32Error());
    }

    private static void SetActiveBuffer(SafeFileHandle handle)
    {
        if (!SetConsoleActiveScreenBuffer(handle))
            throw new Win32Exception(Marshal.GetLastWin32Error());
    }

    private static void FillCells(CHAR_INFO[] cells, char character, short attributes)
    {
        for (int i = 0; i < cells.Length; i++)
        {
            cells[i].UnicodeChar = character;
            cells[i].Attributes = attributes;
        }
    }

    private static short MakeAttribute(ConsoleColor foreground, ConsoleColor background)
    {
        int fg = (int)foreground;
        int bg = (int)background;
        if ((fg & ~0xF) != 0 || (bg & ~0xF) != 0)
            throw new ArgumentOutOfRangeException(nameof(foreground), "ConsoleColor должен быть в диапазоне 0..15.");
        return (short)(fg | bg << 4);
    }
}.

﻿// Rendering/FrameExchange.cs
using AsciiFarm.Core;

namespace AsciiFarm.Rendering;

public sealed class FrameExchange<T> where T : class
{
    private readonly object _gate = new();
    private readonly ThreadGuard _writerGuard = new("FrameExchange.Writer");
    private readonly ThreadGuard _readerGuard = new("FrameExchange.Reader");

    private T[] _buffers;
    private int _writeIndex;
    private int _readIndex = 1;
    private int _readyIndex = 2;
    private bool _hasFreshFrame;

    private Func<T>? _pendingResize;   // защищён _gate; применяет только писатель

    public FrameExchange(Func<T> factory) =>
        _buffers = new[] { factory(), factory(), factory() };

    public void AttachWriter() => _writerGuard.AttachToCurrentThread();
    public void AttachReader() => _readerGuard.AttachToCurrentThread();

    public T WriteBuffer
    {
        get
        {
            _writerGuard.Validate();
            ApplyPendingResize();                 // безопасная точка: кадр ещё не начат
            lock (_gate) return _buffers[_writeIndex];
        }
    }

    public void Publish()
    {
        _writerGuard.Validate();
        lock (_gate)
        {
            (_writeIndex, _readyIndex) = (_readyIndex, _writeIndex);
            _hasFreshFrame = true;
        }
    }

    public bool TryAcquireFresh(out T frame)
    {
        _readerGuard.Validate();
        lock (_gate)
        {
            if (_hasFreshFrame)
            {
                (_readIndex, _readyIndex) = (_readyIndex, _readIndex);
                _hasFreshFrame = false;
                frame = _buffers[_readIndex];
                return true;
            }
            frame = _buffers[_readIndex];
            return false;
        }
    }

    /// <summary>Заявка на смену размера (вызывается из render-потока).</summary>
    public void Resize(Func<T> factory)
    {
        lock (_gate) _pendingResize = factory;
    }

    private void ApplyPendingResize()
    {
        Func<T>? factory;
        lock (_gate)
        {
            factory = _pendingResize;
            if (factory == null) return;
            _pendingResize = null;
        }

        // Аллокация вне замка; _buffers меняет только писатель.
        var newBuffers = new[] { factory(), factory(), factory() };

        lock (_gate)
        {
            _buffers = newBuffers;
            _writeIndex = 0;
            _readIndex = 1;
            _readyIndex = 2;
            _hasFreshFrame = false;
        }
    }
}.

public sealed class HudRenderer : IFrameRenderer

    public void Render(RenderFrame frame, World world)
    {
        int totalWidth = ColWidth * 2 + Gap;
        if (frame.Width < totalWidth + 5) return;

        int startX = frame.Width - totalWidth;
        int col1 = startX;
        int col2 = startX + ColWidth + Gap;

        DrawDateTime(frame, col1, 0, world.Resource<GameState>());
        DrawTileInfo(frame, world, col1, Math.Max(0, frame.Height - 12));

    private static void DrawDateTime(RenderFrame f, int x, int y, GameState state)
    {
        string gameTime = CalendarHelper.FormatGameTime(state.GameStartDate, state.GameTimeElapsed, state.DaysPerMonth);
        var parts = gameTime.Split(' ');
        var real = TimeSpan.FromSeconds(state.ElapsedTime);

        FrameDraw.Text(f, x, y++, "── ВРЕМЯ ──", ConsoleColor.Cyan);
        FrameDraw.Text(f, x, y++, parts.Length > 0 ? parts[0] : "", ConsoleColor.White);

    private static void DrawTileInfo(RenderFrame f, World world, int x, int y)
    {
        var players = world.Query<PlayerTag, Position>();
        var maps = world.Query<TileMapComponent>();
        if (players.Length == 0 || maps.Length == 0) return;

using System.Reflection;
using System.Text;
using AsciiFarm.Core;

namespace AsciiFarm.Serialization;

/// <summary>
/// Формат: "AFARM2" | int version | int count | записи { string type, int ver, int id, int len, byte[len] }.
/// </summary>
public sealed class ComponentSaveSystem
{
    private static readonly byte[] Signature = Encoding.ASCII.GetBytes("AFARM2");
    public const int FormatVersion = 1;

    // P0-5: жёсткие лимиты против «злых»/битых файлов
    public const int MaxComponentBytes = 64 * 1024 * 1024;   // 64 МБ на компонент
    public const int MaxRecords = 1_000_000;
    private const int MinRecordBytes = 13;                   // 1 (string) + 4 + 4 + 4

    private readonly ComponentSerializerRegistry _registry;
    public ComponentSaveSystem(ComponentSerializerRegistry registry) => _registry = registry;

    public void Save(string filePath, World world)
    {
        var entries = new List<Entry>();
        var visitor = new SaveVisitor(_registry, entries);

        foreach (var (componentType, store) in world.AllStores())
        {
            if (componentType.GetCustomAttribute<DontSaveAttribute>() != null) continue;
            if (_registry.Get(componentType) == null) continue;
            store.Accept(visitor);      // P2-14: типобезопасный обход, без boxing
        }

        using var stream = new FileStream(filePath, FileMode.Create, FileAccess.Write);
        using var writer = new BinaryWriter(stream, Encoding.UTF8);
        writer.Write(Signature);
        writer.Write(FormatVersion);
        writer.Write(entries.Count);
        foreach (var e in entries)
        {
            writer.Write(e.TypeName);
            writer.Write(e.Version);
            writer.Write(e.EntityId);
            writer.Write(e.Data.Length);
            writer.Write(e.Data);
        }
    }

    public void Load(string filePath, World world)
    {
        using var stream = new FileStream(filePath, FileMode.Open, FileAccess.Read);
        using var reader = new BinaryReader(stream, Encoding.UTF8);

        var sig = reader.ReadBytes(Signature.Length);
        if (!sig.AsSpan().SequenceEqual(Signature))
            throw new InvalidDataException("Неверная сигнатура файла сохранения.");

        int formatVersion = reader.ReadInt32();
        if (formatVersion <= 0 || formatVersion > FormatVersion)
            throw new InvalidDataException($"Формат {formatVersion} не поддерживается (максимум {FormatVersion}).");

        int count = reader.ReadInt32();
        if (count < 0 || count > MaxRecords || count > (stream.Length - stream.Position) / MinRecordBytes)
            throw new InvalidDataException($"Некорректное число записей: {count}.");

        int maxEntityId = -1;
        for (int i = 0; i < count; i++)
        {
            string typeName = reader.ReadString();
            int version = reader.ReadInt32();
            int entityId = reader.ReadInt32();
            int dataLen = reader.ReadInt32();

            if (entityId < 0)
                throw new InvalidDataException($"Некорректный ID сущности: {entityId}.");
            if (dataLen < 0 || dataLen > MaxComponentBytes)
                throw new InvalidDataException($"Компонент '{typeName}': недопустимый размер {dataLen} (лимит {MaxComponentBytes}).");
            if (dataLen > stream.Length - stream.Position)
                throw new InvalidDataException($"Компонент '{typeName}': файл оборван.");

            byte[] data = reader.ReadBytes(dataLen);
            if (data.Length != dataLen) throw new EndOfStreamException();

            var serializer = _registry.Get(typeName);
            if (serializer == null || version <= 0 || version > serializer.Version)
                continue;   // forward compatibility

            using var ms = new MemoryStream(data, writable: false);
            using var compReader = new BinaryReader(ms, Encoding.UTF8, leaveOpen: true);
            var component = serializer.Deserialize(compReader, version);

            serializer.Applier(world, new Entity(entityId), component);   // P2-16: без рефлексии
            if (entityId > maxEntityId) maxEntityId = entityId;
        }

        if (maxEntityId >= 0) world.ReserveId(maxEntityId);
    }

    public static bool IsNewFormat(string filePath)
    {
        if (!File.Exists(filePath)) return false;
        using var stream = File.OpenRead(filePath);
        if (stream.Length < Signature.Length) return false;
        var sig = new byte[Signature.Length];
        return stream.Read(sig, 0, sig.Length) == sig.Length && sig.AsSpan().SequenceEqual(Signature);
    }

    private readonly record struct Entry(string TypeName, int Version, int EntityId, byte[] Data);

    /// <summary>Обход компонентов без упаковки: T известен статически.</summary>
    private sealed class SaveVisitor : IComponentVisitor
    {
        private readonly ComponentSerializerRegistry _registry;
        private readonly List<Entry> _entries;
        private readonly MemoryStream _ms = new();
        private readonly BinaryWriter _writer;

        public SaveVisitor(ComponentSerializerRegistry registry, List<Entry> entries)
        {
            _registry = registry;
            _entries = entries;
            _writer = new BinaryWriter(_ms, Encoding.UTF8, leaveOpen: true);
        }

        public void Visit<T>(Entity entity, in T component) where T : struct
        {
            var typed = _registry.Get<T>();
            if (typed != null)
            {
                Reset();
                typed.Serialize(_writer, component);
                _writer.Flush();
                _entries.Add(new Entry(typed.TypeName, typed.Version, entity.Id, _ms.ToArray()));
                return;
            }

            // Fallback для сериализаторов, не наследующих ComponentSerializer<T>.
            var untyped = _registry.Get(typeof(T));
            if (untyped == null) return;
            Reset();
            untyped.Serialize(_writer, component);
            _writer.Flush();
            _entries.Add(new Entry(untyped.TypeName, untyped.Version, entity.Id, _ms.ToArray()));
        }

        private void Reset() { _ms.SetLength(0); _ms.Position = 0; }
    }
}.

public sealed class TileMapSerializer : ComponentSerializer<TileMapComponent>
{
    /// <summary>P0-5: лимит тайлов, чтобы битый файл не вызвал OOM при аллокации.</summary>
    public const int MaxTiles = 2_000_000;   // ~36 МБ при 18 байтах на тайл

    public override string TypeName => nameof(TileMapComponent);
    public override int Version => 2;

    public override void Serialize(BinaryWriter w, TileMapComponent m)
    {
        int width = m.Width, height = m.Height;
        w.Write(width);
        w.Write(height);

        for (int y = 0; y < height; y++)
            for (int x = 0; x < width; x++)
            {
                var tile = m.Tiles[x, y];
                w.Write(tile.Symbol);
                w.Write(tile.Nitrogen);
                w.Write(tile.Phosphorus);
                w.Write(tile.Potassium);
                w.Write(tile.Moisture);
            }
    }

    public override TileMapComponent Deserialize(BinaryReader r, int version)
    {
        if (version is < 1 or > 2)
            throw new InvalidDataException($"TileMap: неизвестная версия {version}.");

        int width = r.ReadInt32();
        int height = r.ReadInt32();
        ValidateSize(width, height);          // проверка ДО new TileData[...]

        var tiles = new TileData[width, height];
        for (int y = 0; y < height; y++)
            for (int x = 0; x < width; x++)
            {
                char symbol = r.ReadChar();
                tiles[x, y] = version == 1
                    ? MigrateTile(symbol)
                    : new TileData
                    {
                        Symbol = symbol,
                        Nitrogen = r.ReadSingle(),
                        Phosphorus = r.ReadSingle(),
                        Potassium = r.ReadSingle(),
                        Moisture = r.ReadSingle()
                    };
            }

        return new TileMapComponent { Tiles = tiles };
    }

    private static void ValidateSize(int width, int height)
    {
        if (width <= 0 || height <= 0)
            throw new InvalidDataException($"TileMap: некорректный размер {width}x{height}.");

        if ((long)width * height > MaxTiles)
            throw new InvalidDataException(
                $"TileMap: {width}x{height} превышает лимит {MaxTiles} тайлов.");
    }

    private static TileData MigrateTile(char symbol) => symbol switch
    {
        '≈' => new TileData { Symbol = symbol, Nitrogen = 0f, Phosphorus = 0f, Potassium = 0f, Moisture = 1.0f },
        '♣' or '♠' => new TileData { Symbol = symbol, Nitrogen = 0.7f, Phosphorus = 0.5f, Potassium = 0.6f, Moisture = 0.7f },
        '.' or ',' or '\'' or '`' or ';' => new TileData { Symbol = symbol, Nitrogen = 0.5f, Phosphorus = 0.3f, Potassium = 0.4f, Moisture = 0.6f },
        '#' or '▒' or '┌' or '┐' or '└' or '┘' or '─' or '│' or '┬' or '┴' or '├' or '┤' or '┼' or '╪'
            or '╔' or '╗' or '╚' or '╝' or '═' or '║' or '╠' or '╣' or '╦' or '╩' or '╬'
            => new TileData { Symbol = symbol, Nitrogen = 0f, Phosphorus = 0f, Potassium = 0f, Moisture = 0f },
        ' ' => new TileData { Symbol = symbol, Nitrogen = 0.1f, Phosphorus = 0.1f, Potassium = 0.1f, Moisture = 0.1f },
        _ => new TileData { Symbol = symbol, Nitrogen = 0.3f, Phosphorus = 0.3f, Potassium = 0.3f, Moisture = 0.4f }
    };
}

﻿using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using System.Text;
namespace AsciiFarm.Serialization
{
    public class GameStorage
    {
        private static readonly byte[] BinarySignature = Encoding.ASCII.GetBytes("AFARM");
        private readonly Dictionary<Version, IVersionFormatHandler> _handlers = new();
        private readonly Dictionary<Version, IVersionConverter> _converters = new();

        public static Version LatestVersion => new Version(0, 0, 3);

        public GameStorage()
        {
            RegisterHandler(new FormatHandlerV0_0_0());
            RegisterHandler(new FormatHandlerV0_0_1());
            RegisterHandler(new FormatHandlerV0_0_2());
            RegisterHandler(new FormatHandlerV0_0_3());

            RegisterConverter(new ConverterV0_0_0ToV0_0_1());
            RegisterConverter(new ConverterV0_0_1ToV0_0_2());
            RegisterConverter(new ConverterV0_0_2ToV0_0_3());
            RegisterConverter(new ConverterV0_0_3ToV0_0_2());
        }

        public void RegisterHandler(IVersionFormatHandler handler) =>
            _handlers[handler.Version] = handler;

        public void RegisterConverter(IVersionConverter converter) =>
            _converters[converter.SourceVersion] = converter;

        public void Save(string filePath, Version version, object data)
        {
            if (!_handlers.TryGetValue(version, out var handler))
                throw new NotSupportedException($"Обработчик для версии {version} не найден.");
            using var stream = new FileStream(filePath, FileMode.Create, FileAccess.Write);
            if (version > new Version(0, 0, 0))
            {
                using var headerWriter = new BinaryWriter(stream, Encoding.UTF8, leaveOpen: true);
                headerWriter.Write(BinarySignature);
                headerWriter.Write(version.Major);
                headerWriter.Write(version.Minor);
                headerWriter.Write(version.Build);
            }
            handler.Save(stream, data);
        }

        public (Version Version, object Data) Load(string filePath)
        {
            using var stream = new FileStream(filePath, FileMode.Open, FileAccess.Read);
            Version detectedVersion;
            if (HasBinarySignature(stream))
            {
                using var headerReader = new BinaryReader(stream, Encoding.UTF8, leaveOpen: true);
                int major = headerReader.ReadInt32();
                int minor = headerReader.ReadInt32();
                int build = headerReader.ReadInt32();
                detectedVersion = new Version(major, minor, build);
            }
            else
            {
                stream.Position = 0;
                detectedVersion = new Version(0, 0, 0);
            }
            if (!_handlers.TryGetValue(detectedVersion, out var handler))
                throw new NotSupportedException($"Неизвестная версия сохранения: {detectedVersion}");
            object data = handler.Load(stream);
            return (detectedVersion, data);
        }

        public object ConvertStep(Version currentVersion, object data)
        {
            if (!_converters.TryGetValue(currentVersion, out var converter))
                throw new InvalidOperationException($"Нет конвертера для перехода с версии {currentVersion}.");
            return converter.Convert(data);
        }

        public T LoadAndConvert<T>(string filePath, Version targetVersion) where T : class
        {
            var (currentVersion, currentData) = Load(filePath);
            while (currentVersion < targetVersion)
            {
                if (!_converters.TryGetValue(currentVersion, out var converter))
                    throw new InvalidOperationException($"Не найден конвертер для версии {currentVersion}.");
                currentData = converter.Convert(currentData);
                currentVersion = converter.TargetVersion;
            }
            if (currentVersion != targetVersion)
                throw new InvalidOperationException($"Версия файла ({currentVersion}) выше целевой ({targetVersion}).");
            return currentData as T
                ?? throw new InvalidCastException($"Данные версии {targetVersion} не соответствуют типу {typeof(T).Name}.");
        }

        public T LoadLatest<T>(string filePath) where T : class
        {
            return LoadAndConvert<T>(filePath, LatestVersion);
        }

        /// <summary>
        /// Сохраняет данные. Если версия файла совпадает — перезаписывает.
        /// Если файл старее — конвертирует прочитанное и перезаписывает тем же путём.
        /// Если файл новее — создаёт новый файл с суффиксом версии.
        /// </summary>
        public void SaveSmart(string filePath, Version version, object data)
        {
            if (!File.Exists(filePath))
            {
                Save(filePath, version, data);
                return;
            }

            var (existingVersion, _) = Load(filePath);

            if (existingVersion == version)
            {
                // Та же версия — просто перезаписываем
                Save(filePath, version, data);
            }
            else if (existingVersion < version)
            {
                // Файл старее: перезаписываем его новой версией
                Save(filePath, version, data);
            }
            else
            {
                // Файл новее — создаём отдельный файл с суффиксом
                string dir = Path.GetDirectoryName(filePath) ?? ".";
                string name = Path.GetFileNameWithoutExtension(filePath);
                string ext = Path.GetExtension(filePath);
                string newFilePath = Path.Combine(dir, $"{name}_v{version.Major}.{version.Minor}.{version.Build}{ext}");
                Save(newFilePath, version, data);
            }
        }

        private static bool HasBinarySignature(Stream stream)
        {
            if (stream.Length < BinarySignature.Length)
                return false;
            byte[] buffer = new byte[BinarySignature.Length];
            int bytesRead = stream.Read(buffer, 0, buffer.Length);
            return bytesRead == BinarySignature.Length && buffer.SequenceEqual(BinarySignature);
        }
    }

    public class FormatHandlerV0_0_0 : IVersionFormatHandler
    {
        public Version Version => new Version(0, 0, 0);

        public void Save(Stream stream, object data)
        {
            if (data is not GameDataV0_0_0 state)
                throw new ArgumentException($"Ожидался тип {nameof(GameDataV0_0_0)}");
            using var writer = new StreamWriter(stream, Encoding.UTF8, leaveOpen: true);
            int width = state.Map.GetLength(0);
            int height = state.Map.GetLength(1);
            for (int y = 0; y < height; y++)
            {
                char[] row = new char[width];
                for (int x = 0; x < width; x++) row[x] = state.Map[x, y];
                writer.WriteLine(row);
            }
        }

        public object Load(Stream stream)
        {
            using var reader = new StreamReader(stream, Encoding.UTF8, leaveOpen: true);
            string content = reader.ReadToEnd();
            string[] lines = content.Split(new[] { "\r\n", "\n", "\r" }, StringSplitOptions.None);
            int height = lines.Length;
            while (height > 0 && string.IsNullOrEmpty(lines[height - 1])) height--;
            if (height == 0) return new GameDataV0_0_0 { Map = new char[0, 0] };
            int width = 0;
            for (int y = 0; y < height; y++)
                if (lines[y].Length > width) width = lines[y].Length;
            char[,] map = new char[width, height];

using AsciiFarm.Data;
using AsciiFarm.Rendering;
using System.Diagnostics;

namespace AsciiFarm.Systems;

public sealed class ConsolePresentSystem
{
    private readonly DoubleBufferedConsole _console;
    private readonly FrameExchange<RenderFrame> _frames;
    private readonly FpsCounter _fpsCounter;
    private readonly Stopwatch _clock = Stopwatch.StartNew();   // P0-3: реальное время

    public ConsolePresentSystem(DoubleBufferedConsole console,
                                FrameExchange<RenderFrame> frames,
                                FpsCounter fpsCounter)
    {
        _console = console;
        _frames = frames;
        _fpsCounter = fpsCounter;
    }

    public void Present()
    {
        if (_frames.TryAcquireFresh(out var frame))
        {
            for (int y = 0; y < frame.Height; y++)
                for (int x = 0; x < frame.Width; x++)
                {
                    int i = y * frame.Width + x;
                    _console.PutChar(x, y, frame.Symbols[i], frame.Foreground[i], frame.Background[i]);
                }

            _fpsCounter.RecordFrame(_clock.Elapsed.TotalSeconds);  // не frame.Tick!
        }
        _console.Draw();
    }
}.

[GameSystem(Order = 1)]
public sealed class ElapsedTimeSystem : ISystem   // P2-10: stateless
{
    public void Update(World world)
    {
        var state = world.Resource<GameState>();
        var clock = world.Resource<SimulationClock>();
        var fpsCounter = world.Resource<FpsCounter>();

        double realDeltaTime = Math.Min(clock.NextDelta(), 0.1);

        state.ElapsedTime += realDeltaTime;
        state.DeltaTime = (float)realDeltaTime;
        state.Tick++;                       // единственный инкремент Tick в проекте
        fpsCounter.RecordTick();

        if (state.FastForwardUntil.HasValue) return;   // время уже добавлено в SimulationLoop

        state.GameTimeElapsed += Math.Min(realDeltaTime * state.TimeSpeed, state.MaxTimeStep);
    }
}.

using AsciiFarm.Components;
using AsciiFarm.Core;
using AsciiFarm.Data;
namespace AsciiFarm.Systems;

/// <summary>Бывший MapPhysics: проходимость берётся из данных палитры, а не из switch.</summary>
[GameSystem(Order = 4)]
public sealed class MovementSystem : ISystem
{
    public void Update(World world)
    {
        var maps = world.Query<TileMapComponent>();
        if (maps.Length == 0) return;

        var map = world.Store<TileMapComponent>().Get(maps[0]);
        var palette = world.Resource<TilePalette>();
        var positions = world.Store<Position>();
        var intents = world.Store<MoveIntent>();

        foreach (var entity in world.Query<Position, MoveIntent>())
        {
            var intent = intents.Get(entity);
            var position = positions.Get(entity);
            int targetX = position.X + intent.Dx;
            int targetY = position.Y + intent.Dy;

            if (IsWalkable(map, palette, targetX, targetY))
                positions.Set(entity, new Position(targetX, targetY));

            intents.Remove(entity);   // намерение одноразовое
        }
    }

    private static bool IsWalkable(TileMapComponent map, TilePalette palette, int x, int y)
    {
        if (x < 0 || y < 0 || x >= map.Width || y >= map.Height) return false;
        return !palette.Resolve(map.Tiles[x, y].Symbol).BlocksMovement;
    }
}.

using AsciiFarm.Components;
using AsciiFarm.Core;
using AsciiFarm.Data;
using AsciiFarm.Input;

namespace AsciiFarm.Systems;

[GameSystem(Order = 3)]
public sealed class PlayerIntentSystem : ISystem
{
    public void Update(World world)
    {
        var commands = world.Resource<CommandBuffer>().Items;
        if (commands.Count == 0) return;

        var state = world.Resource<GameState>();
        var intents = world.Store<MoveIntent>();
        Entity[]? players = null;

        foreach (var command in commands)   // один проход
        {
            switch (command)
            {
                case GameCommand.Exit:
                    state.IsRunning = false;
                    state.NeedsSave = true;
                    break;

                case GameCommand.MoveUp: Move(0, -1); break;
                case GameCommand.MoveDown: Move(0, 1); break;
                case GameCommand.MoveLeft: Move(-1, 0); break;
                case GameCommand.MoveRight: Move(1, 0); break;

                // Относительные скорости
                case GameCommand.Speed1: state.TimeSpeed = state.DefaultTimeSpeed; break;
                case GameCommand.Speed2: state.TimeSpeed = state.DefaultTimeSpeed * 10; break;

                // Абсолютные: сутки / месяц / год ИГРОВОГО календаря за реальную минуту
                case GameCommand.Speed3:
                    state.TimeSpeed = CalendarHelper.SecondsPerDay / CalendarHelper.SecondsPerMinute;
                    break;
                case GameCommand.Speed4:
                    state.TimeSpeed = MonthSeconds(state) / CalendarHelper.SecondsPerMinute;
                    break;
                case GameCommand.Speed5:
                    state.TimeSpeed = YearSeconds(state) / CalendarHelper.SecondsPerMinute;
                    break;

                // Промотка: час / сутки / неделя / месяц / год ИГРОВОГО календаря
                case GameCommand.FastForward1: Forward(CalendarHelper.SecondsPerHour); break;
                case GameCommand.FastForward2: Forward(CalendarHelper.SecondsPerDay); break;
                case GameCommand.FastForward3: Forward(7 * CalendarHelper.SecondsPerDay); break;
                case GameCommand.FastForward4: Forward(MonthSeconds(state)); break;
                case GameCommand.FastForward5: Forward(YearSeconds(state)); break;
            }
        }

        commands.Clear();

        void Move(int dx, int dy)
        {
            players ??= world.Query<PlayerTag, Position>();
            foreach (var entity in players) intents.Set(entity, new MoveIntent(dx, dy));
        }

        void Forward(double seconds) => state.FastForwardUntil = state.GameTimeElapsed + seconds;
    }

    private static double MonthSeconds(GameState s) =>
        CalendarHelper.GetMonthSeconds(s.GameStartDate, s.GameTimeElapsed, s.DaysPerMonth);

    private static double YearSeconds(GameState s) =>
        CalendarHelper.GetYearSeconds(s.GameStartDate, s.GameTimeElapsed, s.DaysPerMonth);
}.

using AsciiFarm.Components;
using AsciiFarm.Core;
using AsciiFarm.Data;

namespace AsciiFarm.Systems;

[GameSystem(Order = 5)]
public sealed class RiverFlowSystem : ISystem
{
    private const char WaterSymbol = '≈';
    private const float AnimationSpeed = 1f;
    private const float BlueProbability = 0.5f;

    public void Update(World world)
    {
        var cache = world.Resource<WaterColorCache>();
        var state = world.Resource<GameState>();

        int bucket = (int)(state.ElapsedTime * AnimationSpeed);
        if (bucket == cache.LastBucket) return;   // картинка не изменилась — не считаем
        cache.LastBucket = bucket;
        cache.Colors.Clear();                     // словарь переиспользуется

        var maps = world.Query<TileMapComponent>();
        if (maps.Length == 0) return;
        var map = world.Store<TileMapComponent>().Get(maps[0]);

        for (int y = 0; y < map.Height; y++)
            for (int x = 0; x < map.Width; x++)
            {
                if (map.Tiles[x, y].Symbol != WaterSymbol) continue;
                int hash = HashCode.Combine(x, y, bucket);
                float value = (hash & 0xFFFF) / (float)0xFFFF;
                var fg = value < BlueProbability ? ConsoleColor.Cyan : ConsoleColor.White;
                cache.Colors[(x, y)] = new TileStyle(fg, ConsoleColor.Black, false);
            }
    }
}.

using System;

using System.Reflection;
using AsciiFarm.Core;
namespace AsciiFarm.Systems;

public static class SystemPipeline
{
    public static ISystem[] CreateSimulationSystems()
    {
        return Assembly.GetExecutingAssembly()
            .GetTypes()
            .Where(t => typeof(ISystem).IsAssignableFrom(t) && !t.IsAbstract && !t.IsInterface)
            .Select(t => new { Type = t, Attr = t.GetCustomAttribute<GameSystemAttribute>() })
            .Where(x => x.Attr != null)
            .OrderBy(x => x.Attr.Order)
            .Select(x => (ISystem)(Activator.CreateInstance(x.Type) ?? throw new InvalidOperationException($"Не удалось создать экземпляр {x.Type.Name}")))
            .ToArray();
    }
}.

        weather.HasData = true;

        sys.LastWeatherIndex = index;
    }
